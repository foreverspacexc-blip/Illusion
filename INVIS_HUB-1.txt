-- helper
local Connections = {}
local function connect(signal, fn)
	local c = signal:Connect(fn)
	table.insert(Connections, c)
	return c
end

local function create(class, props, children)
	local o = Instance.new(class)
	for k, v in pairs(props or {}) do
		if k ~= "Parent" then o[k] = v end
	end
	for _, c in ipairs(children or {}) do c.Parent = o end
	if props and props.Parent then o.Parent = props.Parent end
	return o
end

local function tween(o, t, p, style, dir)
	local tw = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), p)
	tw:Play()
	return tw
end

local Reg, spinList = {}, {}
local function onTheme(fn) table.insert(Reg, fn); fn(Theme, true) end
local function bind(obj, prop, key)
	onTheme(function(t, init)
		if init then obj[prop] = t[key] else tween(obj, .35, {[prop] = t[key]}) end
	end)
end
local function gradAccent(g)
	onTheme(function(t) g.Color = ColorSequence.new(t.Accent, t.Accent2) end)
end
local function SetTheme(name)
	if not Themes[name] then return end
	ThemeName, Theme = name, Themes[name]
	for _, fn in ipairs(Reg) do fn(Theme, false) end
end

-- ระบบแท็บ
local Tabs, CurrentTab = {}, nil

local function paintTab(tab)
	local active = CurrentTab == tab
	tween(tab.Button, .25, {BackgroundColor3 = Theme.Element, BackgroundTransparency = active and .05 or 1})
	tween(tab.Bar, .25, {
		BackgroundColor3 = Theme.Accent2, BackgroundTransparency = active and 0 or 1,
		Size = active and UDim2.fromOffset(3, 18) or UDim2.fromOffset(3, 0),
	})
	tween(tab.Label, .25, {TextColor3 = active and Theme.Text or Theme.SubText})
end

local function selectTab(tab)
	if CurrentTab == tab then return end
	CurrentTab = tab
	for _, t in ipairs(Tabs) do
		t.Page.Visible = (t == tab)
		paintTab(t)
	end
	tab.Page.Position = UDim2.fromOffset(0, 18)
	tween(tab.Page, .4, {Position = UDim2.fromOffset(0, 0)})
end

local function makeBox(page, h)
	local f = create("Frame", {
		Size = UDim2.new(1, 0, 0, h), BackgroundTransparency = .2, BorderSizePixel = 0, Parent = page,
	}, {corner(8)})
	bind(f, "BackgroundColor3", "Element")
	local s = create("UIStroke", {Thickness = 1, Transparency = .75, Parent = f})
	bind(s, "Color", "Accent")
	f.MouseEnter:Connect(function() tween(s, .2, {Transparency = .3}) end)
	f.MouseLeave:Connect(function() tween(s, .2, {Transparency = .75}) end)
	return f
end

local function label(parent, props, key)
	props.BackgroundTransparency = 1
	props.Parent = parent
	props.Font = props.Font or Enum.Font.GothamMedium
	local l = create("TextLabel", props)
	bind(l, "TextColor3", key or "Text")
	return l
end

		if CurrentTab ~= tab then tween(tab.Button, .2, {BackgroundTransparency = .7, BackgroundColor3 = Theme.Element}) end

	tab.Button.MouseButton1Click:Connect(function() selectTab(tab) end)

		inc = inc or 1
		local f = makeBox(page, 52)
		label(f, {
			Position = UDim2.fromOffset(12, 5), Size = UDim2.new(1, -90, 0, 20), Text = text, TextSize = 13,

			AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -12, 0, 5), Size = UDim2.fromOffset(70, 20),
			Text = "", Font = Enum.Font.GothamBold, TextSize = 13, TextXAlignment = Enum.TextXAlignment.Right,
		}, "Accent2")
		local track = create("Frame", {
			Position = UDim2.new(0, 12, 1, -17), Size = UDim2.new(1, -24, 0, 6), BorderSizePixel = 0, Parent = f,
		}, {corner(3)})
		bind(track, "BackgroundColor3", "Background")
		local fill = create("Frame", {
			Size = UDim2.fromScale(0, 1), BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0, Parent = track,
		}, {corner(3)})
		gradAccent(create("UIGradient", {Parent = fill}))
		create("Frame", {
			AnchorPoint = Vector2.new(.5, .5), Position = UDim2.fromScale(1, .5), Size = UDim2.fromOffset(13, 13),
			BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0, Parent = fill,
		}, {corner(7)})

		local function set(v, fire)
			v = math.clamp(math.floor(v / inc + .5) * inc, min, max)
			v = math.floor(v * 100 + .5) / 100
			tween(fill, .08, {Size = UDim2.fromScale((v - min) / (max - min), 1)})
			val.Text = tostring(v)
			if fire and callback then task.spawn(callback, v) end
		end
		set(default or min, false)

		local dragging = false
		local function fromInput(input)
			local rel = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
			set(min + (max - min) * rel, true)
		end
		local hit = create("TextButton", {
			Position = UDim2.new(0, 0, 1, -32), Size = UDim2.new(1, 0, 0, 30), BackgroundTransparency = 1, Text = "", Parent = f,

			end
		end)
		connect(UIS.InputChanged, function(input)
			if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
				fromInput(input)
			end
		end)
		connect(UIS.InputEnded, function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				dragging = false
				page.ScrollingEnabled = true
			end
		end)
	end

local flyConn, flyBV, flyBG
local function stopFly()
	if flyConn then flyConn:Disconnect(); flyConn = nil end
	if flyBV then flyBV:Destroy(); flyBV = nil end
	if flyBG then flyBG:Destroy(); flyBG = nil end
	local h = getHum()
	if h then h.PlatformStand = false end
end
local function startFly()
	stopFly()
	local hrp, hum = getRoot(), getHum()
	if not hrp or not hum then return end
	flyBV = create("BodyVelocity", {MaxForce = Vector3.one * 1e5, Velocity = Vector3.zero, Parent = hrp})
	flyBG = create("BodyGyro", {MaxTorque = Vector3.one * 1e5, P = 1e4, CFrame = hrp.CFrame, Parent = hrp})
	hum.PlatformStand = true
	flyConn = RunService.RenderStepped:Connect(function()
		local cam = Workspace.CurrentCamera
		local vert = 0
		if UIS:IsKeyDown(Enum.KeyCode.Space) or hum.Jump then vert = 1
		elseif UIS:IsKeyDown(Enum.KeyCode.LeftControl) then vert = -1 end
		flyBV.Velocity = hum.MoveDirection * State.FlySpeed + Vector3.new(0, vert * State.FlySpeed, 0)
		flyBG.CFrame = cam.CFrame
	end)
end
local function setFly(on)
	State.Fly = on
	if on then startFly() else stopFly() end
end

local FarmState = {
    Enabled   = false,
    TargetIdx = 6,      -- workspace.MobFolder:GetChildren()[6]
    Range     = 50,     -- studs — ตรวจก่อนวาร์ป
    HoverDist = 12,     -- studs เหนือ monster
    Delay     = 0.15,   -- วินาทีระหว่างแต่ละ swing
}

local InputEvent = game.ReplicatedStorage
    :WaitForChild("Luka's Additional Remotes")
    :WaitForChild("InputEvent")

local farmConn = nil

local function getRoot()
    local char = LocalPlayer.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function getHum()
    local char = LocalPlayer.Character
    return char and char:FindFirstChildOfClass("Humanoid")
end

-- หาเป้าหมาย monster ตัวที่ 6 ใน MobFolder
local function getTarget()
    local folder = Workspace:FindFirstChild("MobFolder")
    if not folder then return nil end
    local mobs = folder:GetChildren()
    return mobs[FarmState.TargetIdx]
end

-- หา HumanoidRootPart หรือ PrimaryPart ของ monster
local function getMonsterRoot(mob)
    if not mob then return nil end
    local hrp = mob:FindFirstChild("HumanoidRootPart")
        or mob:FindFirstChild("Root")
        or mob:FindFirstChild("Torso")
        or mob.PrimaryPart
    return hrp
end

-- ตรวจว่า monster ยังมีชีวิตอยู่
local function isAlive(mob)
    if not mob or not mob.Parent then return false end
    local hum = mob:FindFirstChildOfClass("Humanoid")
    if hum and hum.Health <= 0 then return false end
    return true
end

-- วาร์ปขึ้นไปอยู่เหนือ monster 12 studs หันหน้าลงมา
local function warpAbove(monsterRoot)
    local hrp = getRoot()
    if not hrp or not monsterRoot then return false end

    local monPos = monsterRoot.Position
    local targetPos = monPos + Vector3.new(0, FarmState.HoverDist, 0)

    -- หันหน้าลงหา monster (lookAt = monPos)
    local lookDir = (monPos - targetPos).Unit
    local cf = CFrame.new(targetPos, targetPos + lookDir)

    hrp.CFrame = cf
    return true
end

-- ดึง tool ดาบใน character
local function getSword()
    local char = LocalPlayer.Character
    if not char then return nil end
    -- หา tool ที่ equipped อยู่ใน character (ไม่ใช่ใน Backpack)
    for _, v in ipairs(char:GetChildren()) do
        if v:IsA("Tool") then
            return v
        end
    end
    return nil
end

-- ยิง M1 attack ผ่าน InputEvent (เหมือน v_u_33 ใน sword script)
local function swingSword(tool, monsterPos)
    if not tool then return end
    -- ใช้ InputEvent:FireServer(tool, mode, mousePos) ตรงๆ
    -- mode = nil สำหรับ M1 default (ดูจาก v_u_33 ใน sword script)
    InputEvent:FireServer(tool, nil, monsterPos)
end

-- loop หลัก
local function farmLoop()
    while FarmState.Enabled do
        local mob = getTarget()

        if not isAlive(mob) then
            -- monster ตายแล้ว รอ respawn
            task.wait(1)
            continue
        end

        local monRoot = getMonsterRoot(mob)
        if not monRoot then
            task.wait(0.5)
            continue
        end

        local hrp = getRoot()
        if not hrp then
            task.wait(0.5)
            continue
        end

        -- ตรวจระยะก่อนวาร์ป — ต้องอยู่ภายใน 50 studs
        local dist = (hrp.Position - monRoot.Position).Magnitude
        if dist <= FarmState.Range then
            -- วาร์ปขึ้นไปอยู่เหนือ monster
            warpAbove(monRoot)
            task.wait(0.05) -- รอให้ CFrame settle

            -- ตีดาบ
            local sword = getSword()
            if sword then
                swingSword(sword, monRoot.Position)
            end
        else
            -- ถ้า monster อยู่ไกลเกิน 50 studs ให้ teleport ตรงๆ ก่อน
            -- (ครั้งแรก หรือ monster วิ่งหนี)
            warpAbove(monRoot)
            task.wait(0.05)
            local sword = getSword()
            if sword then
                swingSword(sword, monRoot.Position)
            end
        end

        task.wait(FarmState.Delay)
    end
end

local function startFarm()
    if farmConn then return end
    FarmState.Enabled = true
    farmConn = task.spawn(farmLoop)
    Notify("Auto Farm", "เริ่ม farming monster #" .. FarmState.TargetIdx, 3)
end

local function stopFarm()
    FarmState.Enabled = false
    farmConn = nil
    Notify("Auto Farm", "หยุด farming", 2)
end

local startTime, frames, acc = os.time(), 0, 0
local function fmtTime(s)
	return string.format("%02d:%02d:%02d", math.floor(s / 3600), math.floor(s % 3600 / 60), s % 60)
end
connect(RunService.RenderStepped, function(dt)
	if State.Animated then
		StrokeGrad.Rotation = (StrokeGrad.Rotation + dt * 70) % 360
		BgGrad.Rotation = 45 + math.sin(os.clock() * .6) * 30
		for _, s in ipairs(spinList) do
			s[1].Rotation = (s[1].Rotation + dt * s[2]) % 360
		end
	end
	frames += 1
	acc += dt
	if acc >= .5 then
		cards.FPS.Text = tostring(math.floor(frames / acc + .5))
		local ok, ping = pcall(function() return math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) end)
		cards.Ping.Text = ok and (ping .. " ms") or "N/A"
		cards.Players.Text = #Players:GetPlayers() .. " / " .. Players.MaxPlayers
		cards.Session.Text = fmtTime(os.time() - startTime)
		frames, acc = 0, 0
	end
end)
