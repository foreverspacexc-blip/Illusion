local function getHRP()
    local character = LocalPlayer.Character
    return character and character:FindFirstChild("HumanoidRootPart")
end

local function getHumanoid()
    local character = LocalPlayer.Character
    return character and character:FindFirstChildOfClass("Humanoid")
end

-- ระบบ Noclip
RunService.Stepped:Connect(function()
    if noclipEnabled then
        local character = LocalPlayer.Character
        if character then
            for _, part in ipairs(character:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end
        end
    end
end)

local function teleportInstant(cf)
    local hrp = getHRP()
    if hrp and cf then
        hrp.CFrame = cf
    end
end

local function teleportTween(cf)
    local hrp = getHRP()
    if not hrp or not cf then return end
    local dist = (hrp.Position - cf.Position).Magnitude
    local duration = math.clamp(dist / tweenSpeed, 0.3, 5)
    local tw = TweenService:Create(hrp, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {CFrame = cf})
    tw:Play()
    tw.Completed:Wait()
end

local function teleportTo(cf)
    if useTween then
        teleportTween(cf)
    else
        teleportInstant(cf)
    end
end

local function getMyPlot()
    local myName = LocalPlayer.Name:lower()
    for _, plot in ipairs(Workspace:GetChildren()) do
        if plot.Name:lower():find("plot_") then
            if plot.Name:lower():find(myName) then
                return plot
            end
        end
    end
    local hrp = getHRP()
    if not hrp then return nil end
    local closestPlot = nil
    local minDist = math.huge
    for _, plot in ipairs(Workspace:GetChildren()) do
        if plot.Name:lower():find("plot_") then
            local part = plot:FindFirstChildWhichIsA("BasePart", true)
            if part then
                local dist = (part.Position - hrp.Position).Magnitude
                if dist < minDist then
                    minDist = dist
                    closestPlot = plot
                end
            end
        end
    end
    return closestPlot
end

local function getPlotReturnCFrame()
    local myPlot = getMyPlot()
    if not myPlot then return nil end
    local floor1 = myPlot:FindFirstChild("Floor1")
    if floor1 then
        local part = floor1:FindFirstChildWhichIsA("BasePart", true)
        if part then
            return part.CFrame + Vector3.new(0, 3, 0)
        end
    end
    local part = myPlot:FindFirstChildWhichIsA("BasePart", true)
    if part then
        return part.CFrame + Vector3.new(0, 3, 0)
    end
    return nil
end

local function getStageCFrame(stageName)
    local gameplay = Workspace:FindFirstChild("Gameplay")
    if not gameplay then return nil end
    local itemSpawners = gameplay:FindFirstChild("ItemSpawners")
    if not itemSpawners then return nil end
    local stage = itemSpawners:FindFirstChild(stageName)
    if not stage then return nil end
    local part = stage:FindFirstChildWhichIsA("BasePart", true)
    if part then
        return part.CFrame + Vector3.new(0, 5, 0)
    end
    local ok, pivot = pcall(function() return stage:GetPivot() end)
    if ok and pivot then
        return pivot + Vector3.new(0, 5, 0)
    end
    return nil
end

local function getModelCFrame(model)
    if model:IsA("Model") then
        if model.PrimaryPart then
            return model.PrimaryPart.CFrame
        end
        local part = model:FindFirstChildWhichIsA("BasePart")
        if part then
            return part.CFrame
        end
        return model:GetPivot()
    elseif model:IsA("BasePart") then
        return model.CFrame
    end
    return nil
end

local function getRandomModelFromStage(stageFolder)
    local models = {}
    for _, obj in ipairs(stageFolder:GetChildren()) do
        if obj:IsA("Model") or obj:IsA("BasePart") then
            table.insert(models, obj)
        end
    end
    if #models == 0 then return nil end
    return models[math.random(1, #models)]
end

local function getHighestValueModelFromStage(stageFolder)
    local bestModel = nil
    local highestVal = -1
    
    for _, obj in ipairs(stageFolder:GetChildren()) do
        if obj:IsA("Model") or obj:IsA("BasePart") then
            local val = 0
            val = obj:GetAttribute("Value") or obj:GetAttribute("Price") or obj:GetAttribute("Money") or 0
            
            if val == 0 then
                local valObj = obj:FindFirstChild("Value") or obj:FindFirstChild("Price") or obj:FindFirstChild("Money")
                if valObj and (valObj:IsA("IntValue") or valObj:IsA("NumberValue")) then
                    val = valObj.Value
                end
            end
            
            if val > highestVal then
                highestVal = val
                bestModel = obj
            end
        end
    end
    
    if not bestModel or highestVal <= 0 then
        return getRandomModelFromStage(stageFolder)
    end
    
    return bestModel
end

local function getTargetModel()
    local gameplay = Workspace:FindFirstChild("Gameplay")
    if not gameplay then return nil end
    local itemSpawners = gameplay:FindFirstChild("ItemSpawners")
    if not itemSpawners then return nil end
    
    if selectedStage == "Every Stage" then
        local bestOverallModel = nil
        local highestOverallVal = -1
        
        for _, stageFolder in ipairs(itemSpawners:GetChildren()) do
            for _, obj in ipairs(stageFolder:GetChildren()) do
                if obj:IsA("Model") or obj:IsA("BasePart") then
                    local val = 0
                    val = obj:GetAttribute("Value") or obj:GetAttribute("Price") or obj:GetAttribute("Money") or 0
                    
                    if val == 0 then
                        local valObj = obj:FindFirstChild("Value") or obj:FindFirstChild("Price") or obj:FindFirstChild("Money")
                        if valObj and (valObj:IsA("IntValue") or valObj:IsA("NumberValue")) then
                            val = valObj.Value
                        end
                    end
                    
                    if val > highestOverallVal then
                        highestOverallVal = val
                        bestOverallModel = obj
                    end
                end
            end
        end
        
        if not bestOverallModel or highestOverallVal <= 0 then
            local allStages = itemSpawners:GetChildren()
            if #allStages > 0 then
                local randomStage = allStages[math.random(1, #allStages)]
                return getRandomModelFromStage(randomStage)
            end
        end
        
        return bestOverallModel
    else
        local stage = itemSpawners:FindFirstChild(selectedStage)
        if not stage then return nil end
        return getHighestValueModelFromStage(stage)
    end
end

local function fireNearbyPrompts(range)
    range = range or 10
    local hrp = getHRP()
    if not hrp then return end
    for _, desc in ipairs(Workspace:GetDescendants()) do
        if desc:IsA("ProximityPrompt") then
            local parent = desc.Parent
            if parent and parent:IsA("BasePart") then
                local dist = (parent.Position - hrp.Position).Magnitude
                if dist <= range then
                    pcall(function()
                        fireproximityprompt(desc)
                    end)
                end
            end
        end
    end
end

local function fireTouchInterest(touchPart)
    local hrp = getHRP()
    if not hrp or not touchPart then return end
    pcall(function()
        firetouchinterest(touchPart, hrp, 0)
        task.wait(0.05)
        firetouchinterest(touchPart, hrp, 1)
    end)
end

local function doRebirth()
    pcall(function()
        local event = ReplicatedStorage:WaitForChild("Events"):WaitForChild("RequestRebirth")
        event:FireServer()
    end)
end

local function doBuyEgg()
    pcall(function()
        local args = {selectedEgg, 1}
        ReplicatedStorage:WaitForChild("Events"):WaitForChild("RequestBuyEgg"):InvokeServer(unpack(args))
    end)
end

-- ระบบ Auto Spin Wheel
local function doSpinWheel()
    pcall(function()
        local spinEvent = ReplicatedStorage:FindFirstChild("Events") and ReplicatedStorage.Events:FindFirstChild("RequestSpinWheel") 
            or ReplicatedStorage:FindFirstChild("Events") and ReplicatedStorage.Events:FindFirstChild("SpinWheel")
            or ReplicatedStorage:FindFirstChild("RequestSpinWheel")
        if spinEvent then
            if spinEvent:IsA("RemoteEvent") then
                spinEvent:FireServer()
            elseif spinEvent:IsA("RemoteFunction") then
                spinEvent:InvokeServer()
            end
        end
    end)
end

-- ระบบ Auto Claim Rewards / Daily
local function doClaimRewards()
    pcall(function()
        for _, remote in ipairs(ReplicatedStorage:GetDescendants()) do
            if remote:IsA("RemoteEvent") or remote:IsA("RemoteFunction") then
                local name = remote.Name:lower()
                if name:find("reward") or name:find("claim") or name:find("daily") then
                    if remote:IsA("RemoteEvent") then
                        remote:FireServer()
                    else
                        remote:InvokeServer()
                    end
                end
            end
        end
    end)
end

local function startAutoCollect()
    task.spawn(function()
        while autoCollectEnabled do
            local myPlot = getMyPlot()
            if myPlot then
                for _, floor in ipairs(myPlot:GetChildren()) do
                    if floor.Name:find("Floor") then
                        local slotsFolder = floor:FindFirstChild("Slots")
                        if slotsFolder then
                            for _, slot in ipairs(slotsFolder:GetChildren()) do
                                if slot.Name:find("Slot") then
                                    local collectTouch = slot:FindFirstChild("CollectTouch")
                                    if collectTouch then
                                        if collectTouch:IsA("BasePart") then
                                            fireTouchInterest(collectTouch)
                                        else
                                            local touchPart = collectTouch:FindFirstChildWhichIsA("BasePart")
                                            if touchPart then
                                                fireTouchInterest(touchPart)
                                            end
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
            task.wait(0.5)
        end
    end)
end

local function startAutoAnimal()
    task.spawn(function()
        while autoAnimalEnabled do
            local model = getTargetModel()
            if model then
                local modelCF = getModelCFrame(model)
                if modelCF then
                    teleportTo(modelCF + Vector3.new(0, 3, 0))
                    task.wait(animalWaitTime)
                    fireNearbyPrompts(12)
                    task.wait(0.2)
                    fireNearbyPrompts(12)
                    task.wait(0.3)
                    
                    local plotCF = getPlotReturnCFrame()
                    if plotCF then
                        teleportTo(plotCF)
                    end
                    task.wait(0.4)
                else
                    task.wait(0.8)
                end
            else
                task.wait(1.5)
            end
        end
    end)
end

local function startAutoRebirth()
    task.spawn(function()
        while autoRebirthEnabled do
            doRebirth()
            task.wait(3)
        end
    end)
end

local function startAutoBuyEgg()
    task.spawn(function()

    end)
end

local function startAutoWheel()
