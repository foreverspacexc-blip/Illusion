-- helper
local Connections = {}
local function connect(signal, fn)
	local c = signal:Connect(fn)
	table.insert(Connections, c)
	return c
end

local function create(class, props, children)
	local o = Instance.new(class)
	for k, v in pairs(props or {}) do
		if k ~= "Parent" then o[k] = v end
	end
	for _, c in ipairs(children or {}) do c.Parent = o end
	if props and props.Parent then o.Parent = props.Parent end
	return o
end

local function tween(o, t, p, style, dir)
	local tw = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), p)
	tw:Play()
	return tw
end

local Reg, spinList = {}, {}
local function onTheme(fn) table.insert(Reg, fn); fn(Theme, true) end
local function bind(obj, prop, key)
	onTheme(function(t, init)
		if init then obj[prop] = t[key] else tween(obj, .35, {[prop] = t[key]}) end
	end)
end
local function gradAccent(g)
	onTheme(function(t) g.Color = ColorSequence.new(t.Accent, t.Accent2) end)
end
local function SetTheme(name)
	if not Themes[name] then return end
	ThemeName, Theme = name, Themes[name]
	for _, fn in ipairs(Reg) do fn(Theme, false) end
end

local Tabs, CurrentTab = {}, nil

local function paintTab(tab)
	local active = CurrentTab == tab
	tween(tab.Button, .25, {BackgroundColor3 = Theme.Element, BackgroundTransparency = active and .05 or 1})
	tween(tab.Bar, .25, {
		BackgroundColor3 = Theme.Accent2, BackgroundTransparency = active and 0 or 1,
		Size = active and UDim2.fromOffset(3, 18) or UDim2.fromOffset(3, 0),
	})
	tween(tab.Label, .25, {TextColor3 = active and Theme.Text or Theme.SubText})
end

local function selectTab(tab)
	if CurrentTab == tab then return end
	CurrentTab = tab
	for _, t in ipairs(Tabs) do
		if t.Page then
			t.Page.Visible = (t == tab)
		end
		paintTab(t)
	end
	if tab and tab.Page then
		tab.Page.Position = UDim2.fromOffset(0, 18)
		tween(tab.Page, .4, {Position = UDim2.fromOffset(0, 0)})
	end
end

	f.MouseLeave:Connect(function() tween(s, .2, {Transparency = .75}) end)
	return f
end

			v = math.clamp(math.floor(v / inc + .5) * inc, min, max)
			v = math.floor(v * 100 + .5) / 100
			tween(fill, .08, {Size = UDim2.fromScale((v - min) / (max - min), 1)})
			val.Text = tostring(v)
			if fire and callback then task.spawn(callback, v) end
					-- ค้นหาฟังก์ชันย่อยชื่อ local function set(v, fire) ด้านใน AddSlider แล้วเพิ่มบรรทัดนี้ลงไปท้ายสุด:
		if fire and SaveConfig then pcall(SaveConfig) end
		
		end
		set(default or min, false)

		local dragging = false
		local function fromInput(input)
			local rel = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
			set(min + (max - min) * rel, true)

			end
		end)
		connect(UIS.InputEnded, function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				dragging = false
				page.ScrollingEnabled = true
			end

        if not zoomies then return false end

        -- VectorVelocity was renamed to Velocity in newer BF versions
        velocity = zoomies.VectorVelocity or zoomies.Velocity
        if not velocity then return false end
        ball_pos = GetBallPos(ball)
        offset = hrp.Position - ball_pos
        speed = velocity.Magnitude
        distance = offset.Magnitude
        if speed < 0.001 or distance < 0.001 then return false end
        
        if not shared._TargetStartedDist[ball] then
            -- Adaptive threshold:
            --   Base = distance/3 catches mid-flight curve early enough at low ping.
            --   But at high ping the server hasn't registered our jump yet when the ball
            --   is already under our feet on the client — so we add a ping-based safety
            --   buffer.  The buffer shrinks to zero when ping is ≤30ms so low-ping players
            --   are not affected.
            local pingMs  = GetPing() or 30
            local pingBuf = math.clamp((pingMs - 30) / 1000 * speed, 0, 8)
            shared._TargetStartedDist[ball] = math.max(distance / 3 - pingBuf, 4)
        end
        if distance <= shared._TargetStartedDist[ball] then
            -- Extra guard: if the ball is physically below or level with us right now
            -- (server hasn't caught up to our jump), do NOT return false — allow parry.
            -- Only block when the ball is still clearly above us.
            if hrp.Position.Y <= ball_pos.Y + 0.5 then
                -- ball is at foot level or below — we're mid-jump, server is behind.
                -- Clear the threshold so next frame re-evaluates fresh.
                shared._TargetStartedDist[ball] = nil
            else
                return false
            end
        end
        
        if hrp.Position.Y > ball_pos.Y + 1 then return false end

        ball_dir = velocity.Unit
        dir_to_me = offset.Unit
        dot = dir_to_me:Dot(ball_dir)
        
        if dot > 0.81238 or (dot > 0.75798 and distance <= 26.4567) then
            return false
        end

        cd_data = GetCurveData(ball)
        ping_val = GetPing()
        speed_t = math.min(speed / 100, 40)
        dir_diff = ball_dir - velocity
        dot_diff = dir_diff.Magnitude > 0.001 and math.clamp(dot - dir_to_me:Dot(dir_diff.Unit), -1, 1) or dot
        dot_thresh = 0.5 - ping_val / 1000
        angle_t = 20 * math.max(dot, 0)
        reach_time = math.max(0, distance / speed - ping_val / 1000)
        dist_thresh = 15 - math.min(distance / 1000, 15) + angle_t + speed_t
        radians = math.rad(math.asin(math.clamp(dot, -1, 1)))
        time_now = os.clock()

        if dot == nil then return false end
        if math.sign(dot) ~= math.sign(cd_data.Dot) then
            if time_now - cd_data.Flip_Time < 0.2 then cd_data.Flips = cd_data.Flips + 1 else cd_data.Flips = 0 end
            cd_data.Flip_Time = time_now
        end
        cd_data.Dot = dot
        cd_data.Lerp_Radians = cd_data.Lerp_Radians + (radians - cd_data.Lerp_Radians) * 0.8

        if speed > 100 and reach_time > ping_val / 10 then
            dist_thresh = math.max(dist_thresh - 15, 15)
        end
        if cd_data.Flips >= 5 then return false end
        if dot < 0 then return true end
        if distance < dist_thresh then return false end
        if dot_diff < dot_thresh then return true end
        if cd_data.Lerp_Radians < 0.018 or dot > 0.81238 then cd_data.Last_Warping = tick() end
        if tick() - cd_data.Last_Warping < reach_time / 1.5 then return true end
        if distance > dist_thresh and tick() - cd_data.Curving < reach_time / 1.5 then return true end
        return dot < dot_thresh
    end)
    if not status then
        print("Illusion Engine Error: IsCurved processing execution failed due to " .. tostring(err))
        return false
    end
    return err
end

local ForceTargetPlayer = nil
local LastClickTime     = 0

local function ForceTarget()
    local alive = workspace:FindFirstChild("Alive")
    local char  = GetChar()
    local hrp   = GetHRP(char)
    if not alive or not hrp then return nil end

    -- Double Click: aim at double-clicked player
    if Cfg.TargetMode == "Double Click" and ForceTargetPlayer then
        local tc = alive:FindFirstChild(ForceTargetPlayer)
        if tc and tc ~= char then return tc end
    end

    -- Nearest: closest enemy
    if Cfg.TargetMode == "Nearest" then
        local best, bestD = nil, math.huge
        for _, c2 in ipairs(alive:GetChildren()) do
            if c2 ~= char and c2.PrimaryPart then
                local d = (hrp.Position - c2.PrimaryPart.Position).Magnitude
                if d < bestD then bestD = d; best = c2 end
            end
        end
        return best
    end

    -- Farest: furthest enemy — aim curve at the player farthest away
    if Cfg.TargetMode == "Farest" then
        local best, worstD = nil, -1
        for _, c2 in ipairs(alive:GetChildren()) do
            if c2 ~= char and c2.PrimaryPart then
                local d = (hrp.Position - c2.PrimaryPart.Position).Magnitude
                if d > worstD then worstD = d; best = c2 end
            end
        end
        return best
    end

    -- Mouse: aim toward wherever the mouse is pointing in world-space
    -- Returns a synthetic "target" struct with a PrimaryPart at mouse hit position
    -- CalcCurveDir reads target.PrimaryPart.Position so we fake it with a CFrame-based part
    if Cfg.TargetMode == "Mouse" then
        local cam = workspace.CurrentCamera
        local mouse = game:GetService("Players").LocalPlayer:GetMouse()
        if cam and mouse then
            local unitRay = cam:ScreenPointToRay(mouse.X, mouse.Y)
            local raycastParams = RaycastParams.new()
            raycastParams.FilterType = Enum.RaycastFilterType.Exclude
            -- exclude local character so ray doesn't hit self
            local myChar = GetChar()
            if myChar then
                raycastParams.FilterDescendantsInstances = {myChar}
            end
            local result = workspace:Raycast(unitRay.Origin, unitRay.Direction * 2000, raycastParams)
            local hitPos = result and result.Position or (unitRay.Origin + unitRay.Direction * 500)
            -- Return a fake target table that CalcCurveDir can read
            return { PrimaryPart = { Position = hitPos } }
        end
        return nil
    end

    return nil
end

-- Curve direction calculation
function CalcCurveDir(ball, target)
    camera = workspace.CurrentCamera
    char = game.Players.LocalPlayer.Character
    hrp = char and char:FindFirstChild("HumanoidRootPart")
    currentMode = Cfg.CurveMode or "Camera"
    lookV = camera.CFrame.LookVector
    upV = camera.CFrame.UpVector
    rightV = camera.CFrame.RightVector
    
    if currentMode == "Camera" then
        return lookV
    elseif currentMode == "Left" then
        return (-rightV * 150000 + (lookV * 0.001)).Unit
    elseif currentMode == "Right" then
        return (rightV * 150000 + (lookV * 0.001)).Unit
    elseif currentMode == "Up" then
        return (upV * 85000 + (lookV * 0.001)).Unit
    elseif currentMode == "Down" then
        return (-upV * 85000 + (lookV * 0.001)).Unit
    elseif currentMode == "Straight" then
        if hrp then
            return (hrp.CFrame.LookVector * 250000).Unit
        end
        return lookV
    elseif currentMode == "Random" then
        randomSpread = Vector3.new(math.random(-50000, 50000), math.random(-25000, 75000), math.random(-50000, 50000)).Unit
        return (lookV + (randomSpread * 350000)).Unit
    elseif currentMode == "Super Curve Random" then
        choices = {-rightV * 450000, rightV * 450000, upV * 250000, -upV * 250000}
        picked = choices[math.random(1, #choices)]
        return (picked + (lookV * 0.001)).Unit
    end
    
    xpcall(function()
        if ball and target and target.PrimaryPart then
            return ((target.PrimaryPart.Position - ball.Position).Unit * 500000).Unit
        end
    end, function(err)
        print("Illusion Engine Error: Target curve trajectory extreme calculation failed due to " .. tostring(err))
    end)
    
    return lookV
end

-- Build snap map
local function BuildSnap()
    local cam   = workspace.CurrentCamera
    local snap  = {}
    local alive = workspace:FindFirstChild("Alive")
    if alive and cam then
        for _, c2 in ipairs(alive:GetChildren()) do
            local p2 = c2.PrimaryPart or c2:FindFirstChild("HumanoidRootPart")
            if p2 then
                local sp = cam:WorldToScreenPoint(p2.Position)
                if sp.Z > 0 then snap[c2.Name] = sp end
            end
        end
    end
    return snap
end

FireParry = function(ball, isSpam)
    cam = workspace.CurrentCamera
    if not cam then 
        print("Illusion Engine Error: CurrentCamera object is not found")
        return 
    end
    target = ForceTarget()
    char = GetChar()
    hrp = GetHRP(char)
    alive = workspace:FindFirstChild("Alive")
    
    curveDirectionVector = CalcCurveDir(ball, target)
    finalCF = CFrame.lookAt(cam.CFrame.Position, cam.CFrame.Position + curveDirectionVector)

    snap = BuildSnap()
    
    -- Always snapshot ALL players — server needs full positional data
    gamePositionsSnapshot = {}
    if cam and alive then
        for _, character in ipairs(alive:GetChildren()) do
            if character:IsA("Model") and character.PrimaryPart then
                screenPos, onScreen = cam:WorldToScreenPoint(character.PrimaryPart.Position)
                gamePositionsSnapshot[character.Name] = screenPos
            end
        end
    end

    -- Determine aim point: Mouse mode = actual mouse screen pos; everything else = viewport center
    local aimPoint
    if Cfg.TargetMode == "Mouse" then
        local mouse = game:GetService("Players").LocalPlayer:GetMouse()
        aimPoint = {mouse.X, mouse.Y}
    else
        aimPoint = {cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2}
    end

    local packetSent = false

    if sendParry and not packetSent then
        local success, err = pcall(function()
            sendParry(finalCF, gamePositionsSnapshot, aimPoint)
        end)
        if success then
            packetSent = true
        else
            print("Illusion Network Error: sendParry path failed execution due to " .. tostring(err))
        end
    end

    if _reverted and next(_reverted) ~= nil and not packetSent then
        local successCount = 0
        for remote, args in pairs(_reverted) do
            if type(args) ~= "table" then continue end
            local success, err = pcall(function()
                pa = {}
                pa[1] = args[1]
                pa[2] = args[2]
                pa[3] = _tokenize and _tokenize(args[2]) or args[3]
                pa[4] = isSpam and 0 or 0.5
                pa[5] = finalCF
                pa[6] = gamePositionsSnapshot
                pa[7] = aimPoint
                pa[8] = false
                if remote:IsA("RemoteEvent") then
                    remote:FireServer(unpack(pa))
                else
                    remote:InvokeServer(unpack(pa))
                end
            end)
            if success then
                successCount = successCount + 1
            else
                print("Illusion Network Error: _reverted remote path failed execution due to " .. tostring(err))
            end
        end
        if successCount > 0 then
            packetSent = true
        end
    end

    if _firstSignal and not packetSent then
        local success, err = pcall(function()
            _firstSignal:Fire(0.5, finalCF, gamePositionsSnapshot, aimPoint, false)
        end)
        if success then
            packetSent = true
        else
            print("Illusion Network Error: _firstSignal path failed execution due to " .. tostring(err))
        end
    end

    if not packetSent then
        pcall(function()
            if mouse1press then mouse1press(); task.wait(0.03); mouse1release() end
            if mouse1click then mouse1click() end
        end)
    end
end

-- NORMAL MODE — FLYTE FORMULA
--=================================================================
local function ParryNormal(ball, distance, speed, velocity, ping_ms)
    local endPosition   = 101 - math.clamp(Cfg.Accuracy, 1, 100)
    local maximumValue  = math.clamp((endPosition - 1) / 99, 0, 1)
    local offsetZ       = maximumValue * maximumValue * (3 - 2 * maximumValue) -- smoothstep
    local openError     = math.clamp(ping_ms / 10, 10, 16.7)
    local step          = math.clamp(ping_ms / 1000 * 0.5 - 0.05, 0, 0.04)
    local clampedY      = math.clamp(Cfg.ParryRange or 0, 0, 10)

    local newDenom  = 2.4 + math.clamp(speed - 9.5, 0, 990.5) * 0.002
    local center    = speed / newDenom * (0.75 + maximumValue * 0.43)
    local isInside  = center <= 9.5
    local clampedX  = isInside
        and math.min(openError, speed * (ping_ms / 1000))
        or  openError + math.min(speed * step, 30)
    local zone = clampedX + math.max(center, 9.5) + clampedY

    -- expanded zone for high accuracy
    if endPosition > 70 then
        local maxLen      = math.clamp((endPosition - 44) / 30, 0, 1)
        local index35     = maxLen * maxLen * (3 - 2 * maxLen)
        local index37     = (isInside and math.min(openError, speed * (ping_ms/1000)) or openError) + math.max(speed / newDenom * (0.75 + maximumValue * 0.2), 9.5)
        local index38     = openError + speed / newDenom * (1.4 + maximumValue * 0.2)
        local clampedDist = math.clamp((speed - 1000) / 800, 0, 1)
        local index39     = index38 + math.max(index37 - index38, 0) * clampedDist
        zone = zone + (index39 + (index37 - index39) * 0 - zone) * index35
    end

    return distance <= zone
end

--=================================================================
-- PRO MODE — BYTE FORMULA
--=================================================================
local function ParryPro(distance, speed, ping_ms)
    local acc = math.clamp(Cfg.Accuracy, 1, 100)
    local SDM
    if acc <= 50 then
        SDM = 0.8 + (acc - 1) * 0.024747474747474751
    else
        SDM = 0.8 + (acc - 1) * 0.0030303030303030303
    end
    local zone = math.clamp(ping_ms / 10, 10, 16) + speed / (2.3497562749127492 + math.clamp(speed, 0, 425) * 0.002) * SDM
    zone = zone + (Cfg.ParryRange or 0)
    return distance <= zone
end

local function CheckPhantomRelease(ball)
    if not PhantomTargetData or PhantomTargetData.Ball ~= ball then return false end
    if os.clock() >= PhantomTargetData.Expires then PhantomTargetData = nil return false end
    
    local enemy = PhantomTargetData.User
    if not enemy or not enemy.Parent then return false end
    
    if tostring(ball:GetAttribute("target") or "") ~= enemy.Name then
        PhantomTargetData.LastPos = nil
        PhantomTargetData.Released = false
        return false
    end
    
    local currentBallPos = GetBallPos(ball)
    if PhantomTargetData.LastPos and currentBallPos ~= PhantomTargetData.LastPos then
        PhantomTargetData.Released = true
    end
    
    PhantomTargetData.LastPos = currentBallPos
    return PhantomTargetData.Released == true

local _flyConn, _flyBV, _flyGyro = nil, nil, nil

local function StartFly()
    local char = GetChar()
    local hrp = GetHRP(char)
    local hum = GetHum(char)
    if not hrp or not hum then return end
    
    hum.PlatformStand = true
    
    if _flyBV then _flyBV:Destroy() end
    if _flyGyro then _flyGyro:Destroy() end
    if _flyConn then _flyConn:Disconnect() end
    
    _flyBV = Instance.new("BodyVelocity")
    _flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    _flyBV.Velocity = Vector3.zero
    _flyBV.Parent = hrp
    
    _flyGyro = Instance.new("BodyGyro")
    _flyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    _flyGyro.P = 9e4
    _flyGyro.Parent = hrp
    
    _flyConn = RunService.RenderStepped:Connect(function()
        if not Cfg.FlyEnabled or not hrp or not hum or not _flyBV or not _flyGyro then return end
        
        hum.PlatformStand = true
        local cam = workspace.CurrentCamera
        if not cam then return end
        
        _flyGyro.CFrame = cam.CFrame
        
        local moveVector = Vector3.zero
        local controlModule = nil
        
        pcall(function()
            controlModule = require(LP.PlayerScripts:FindFirstChild("PlayerModule"):FindFirstChild("ControlModule"))
        end)
        
        if controlModule and controlModule.GetMoveVector then
            local rawVector = controlModule:GetMoveVector()
            moveVector = Vector3.new(rawVector.X, 0, rawVector.Z)
        else
            moveVector = hum.MoveDirection
        end
        
        if moveVector.Magnitude == 0 then
            if UIS:IsKeyDown(Enum.KeyCode.W) then moveVector = moveVector + Vector3.new(0, 0, -1) end
            if UIS:IsKeyDown(Enum.KeyCode.S) then moveVector = moveVector + Vector3.new(0, 0, 1) end
            if UIS:IsKeyDown(Enum.KeyCode.A) then moveVector = moveVector + Vector3.new(-1, 0, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.D) then moveVector = moveVector + Vector3.new(1, 0, 0) end
        end
        
        local sp = Cfg.FlySpeed or 50
        local direction = cam.CFrame:VectorToWorldSpace(moveVector)
        local verticalVel = 0
        
        if UIS:IsKeyDown(Enum.KeyCode.Space) then verticalVel = sp end
        if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then verticalVel = -sp end
        
        _flyBV.Velocity = (direction * sp) + Vector3.new(0, verticalVel, 0)
    end)
end

local function StopFly()
    if _flyConn then _flyConn:Disconnect(); _flyConn = nil end
    if _flyBV then _flyBV:Destroy(); _flyBV = nil end
    if _flyGyro then _flyGyro:Destroy(); _flyGyro = nil end
    
    local hum = GetHum(GetChar())
    if hum then 
        hum.PlatformStand = false 
    end
end

local _orbitAngle = 0
local _orbitConn  = nil
local function SetOrbit(on)
    if _orbitConn then _orbitConn:Disconnect(); _orbitConn = nil end
    if not on then return end
    -- PreSimulation fires before physics step — HRP position is read by the engine immediately,
    -- and setfflag bumps physics sender rate so the server sees the new pos on the very next tick.
    pcall(function() setfflag("S2PhysicsSenderRate", "60") end)
    _orbitConn = RunService.PreSimulation:Connect(function()
        if not Cfg.OrbitEnabled then return end
        local ball = GetBall()
        if not ball or not ball.Parent then return end
        local hrp  = GetHRP(GetChar())
        if not hrp then return end
        local dt = 1 / 60  -- fixed step; PreSimulation doesn't pass dt
        _orbitAngle = _orbitAngle + (Cfg.OrbitSpeed or 5) * dt
        local dist = Cfg.OrbitDistance or 12
        local co, si = math.cos(_orbitAngle) * dist, math.sin(_orbitAngle) * dist
        local targetPos = ball.Position + Vector3.new(co, 0, si)
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.CFrame = CFrame.lookAt(targetPos, ball.Position)
    end)
end

getgenv().swordModel = ""
getgenv().swordAnimations = ""
getgenv().swordFX = ""

task.spawn(function()
    local swordsInst = RS:WaitForChild("Shared", 9e9):WaitForChild("ReplicatedInstances", 9e9):WaitForChild("Swords", 9e9)
    local Swords = require(swordsInst)
    local swordsCtrl = nil
    local hookedFuncs = {}

    task.spawn(function()
        while task.wait() and not swordsCtrl do
            local ok, conns = pcall(getconnections, RS.Remotes.FireSwordInfo.OnClientEvent)
            if ok and conns then
                for _, v in ipairs(conns) do
                    if v.Function and islclosure and islclosure(v.Function) then
                        local ok2, up = pcall(getupvalues, v.Function)
                        if ok2 and #up == 1 and type(up[1]) == "table" then
                            swordsCtrl = up[1]
                            break
                        end
                    end
                end
            end
        end
    end)

    local function getSlashName(name)
        local ok, res = pcall(function() return Swords:GetSword(name) end)
        return (ok and res and res.SlashName) or "SlashEffect"
    end

    local function refreshSlash()
        local targetFX = getgenv().swordFX ~= "" and getgenv().swordFX or getgenv().swordModel
        getgenv().slashName = targetFX ~= "" and getSlashName(targetFX) or "SlashEffect"
    end

    getgenv().updateSword = function()
        if not getgenv().skinChanger then return end
        local char = GetChar()
        if not char then return end
        pcall(function()
            local equipFunc = rawget(Swords, "EquipSwordTo")
            if type(equipFunc) == "function" then
                local ups = getupvalues(equipFunc)
                for i = 1, #ups do
                    if type(ups[i]) == "boolean" then
                        setupvalue(equipFunc, i, false)
                        break
                    end
                end
            end
        end)
        pcall(function() Swords:EquipSwordTo(char, getgenv().swordModel) end)
        task.spawn(function()
            while not swordsCtrl do task.wait(0.5) end
            pcall(function()
                if swordsCtrl.SetSword then
                    swordsCtrl:SetSword(getgenv().swordAnimations ~= "" and getgenv().swordAnimations or getgenv().swordModel)
                end
            end)
            pcall(function()
                local targetFX = getgenv().swordFX ~= "" and getgenv().swordFX or getgenv().swordModel
                if RS.Remotes:FindFirstChild("FireSwordInfo") then
                    RS.Remotes.FireSwordInfo:FireServer(targetFX)
                end
                if swordsCtrl.currentSword ~= nil then swordsCtrl.currentSword = targetFX end
                if swordsCtrl.SwordFX ~= nil then swordsCtrl.SwordFX = targetFX end
            end)
        end)
    end

    task.spawn(function()
        while task.wait(1) do
            local ok, conns = pcall(getconnections, RS.Remotes.ParrySuccessAll.OnClientEvent)
            if ok and type(conns) == "table" then
                for _, v in ipairs(conns) do
                    local fn = v.Function
                    if fn and not hookedFuncs[fn] then
                        hookedFuncs[fn] = true
                        if not (isourclosure and isourclosure(fn)) then
                            v:Disable()
                            local orig = fn
                            RS.Remotes.ParrySuccessAll.OnClientEvent:Connect(function(...)
                                local args = {...}
                                if tostring(args[4]) == LP.Name and getgenv().skinChanger then
                                    refreshSlash()
                                    args[1] = getgenv().slashName
                                    args[3] = getgenv().swordFX ~= "" and getgenv().swordFX or getgenv().swordModel
                                end
                                if setthreadidentity then pcall(setthreadidentity, 2) end
                                pcall(orig, unpack(args))
                            end)
                        end
                    end
                end
            end
        end
    end)

    task.spawn(function()
        while shared._IllusionRunning do
            task.wait(1)
            if getgenv().skinChanger and getgenv().swordModel ~= "" then
                local char = GetChar()
                if char then
                    if LP:GetAttribute("CurrentlyEquippedSword") ~= getgenv().swordModel or not char:FindFirstChild(getgenv().swordModel) then
                        pcall(getgenv().updateSword)
                    end
                    for _, v in ipairs(char:GetChildren()) do
                        if v:IsA("Model") and v.Name ~= getgenv().swordModel and Swords:GetSword(v.Name) then
                            v:Destroy()
                        end
                    end
                end
            end
        end
    end)
end)

local TargetMorphUser = ""
local MorphActiveState = false

local function ApplyAvatarMorph(targetInput)
    if not targetInput or targetInput == "" then return end
    local character = game:GetService("Players").LocalPlayer.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local head = character:FindFirstChild("Head")
    if not humanoid or not head then return end
    
    local num = tonumber(targetInput)
    if not num then
        pcall(function()
            num = game:GetService("Players"):GetUserIdFromNameAsync(targetInput)
        end)
    end
    if not num then return end

    local ok, result = pcall(function()
        return game:GetService("Players"):CreateHumanoidModelFromUserId(num)
    end)

    if ok and result then
        local head2 = result:FindFirstChild("Head")
        local humanoid2 = result:FindFirstChildOfClass("Humanoid")

        if head2 then
            local decal = head:FindFirstChildOfClass("Decal")
            local decal2 = head2:FindFirstChildOfClass("Decal")
            if decal2 then
                if not decal then
                    decal = Instance.new("Decal")
                    decal.Name = "face"
                    decal.Parent = head
                end
                decal.Texture = decal2.Texture
            elseif decal then
                decal:Destroy()
            end

            local specialMesh = head:FindFirstChildOfClass("SpecialMesh")
            local specialMesh2 = head2:FindFirstChildOfClass("SpecialMesh")
            if specialMesh2 then
                if not specialMesh then
                    specialMesh = Instance.new("SpecialMesh")
                    specialMesh.Parent = head
                end
                specialMesh.MeshId = specialMesh2.MeshId
                specialMesh.TextureId = specialMesh2.TextureId
                specialMesh.Scale = specialMesh2.Scale
            end
        end

        for _, child in ipairs(character:GetChildren()) do
            if child:IsA("Accessory") or child:IsA("Clothing") or child:IsA("ShirtGraphic") or child:IsA("BodyColors") or child:IsA("CharacterMesh") then
                child:Destroy()
            end
        end

        local bodyColors = result:FindFirstChildOfClass("BodyColors")
        if bodyColors then bodyColors:Clone().Parent = character end

        if humanoid.RigType == Enum.HumanoidRigType.R15 and humanoid2 then
            for _, v90 in ipairs({ "BodyHeightScale", "BodyWidthScale", "BodyDepthScale", "HeadScale", "ProportionScale" }) do
                local v91 = humanoid2:FindFirstChild(v90)
                local v92 = humanoid:FindFirstChild(v90)
                if v91 and v92 then v92.Value = v91.Value end
            end
        end

        if humanoid.RigType == Enum.HumanoidRigType.R6 then
            for _, child in ipairs(result:GetChildren()) do
                if child:IsA("CharacterMesh") then child:Clone().Parent = character end
            end
        else
            for _, child in ipairs(result:GetChildren()) do
                if child:IsA("MeshPart") then
                    local v90 = character:FindFirstChild(child.Name)
                    if v90 and v90:IsA("MeshPart") then
                        v90.MeshId = child.MeshId
                        v90.TextureID = child.TextureID
                    end
                end
            end
        end

        for _, child in ipairs(result:GetChildren()) do
            if child:IsA("Accessory") then
                local clone = child:Clone()
                clone.Parent = character
                local handle = clone:FindFirstChild("Handle")
                if handle and handle:IsA("BasePart") then
                    handle.CanCollide = false
                    if not handle:FindFirstChildOfClass("WrapLayer") then
                        local attachment = handle:FindFirstChildOfClass("Attachment")
                        local v90 = nil
                        if attachment then
                            for _, child2 in ipairs(character:GetChildren()) do
                                if child2:IsA("BasePart") and child2:FindFirstChild(attachment.Name) then
                                    v90 = child2:FindFirstChild(attachment.Name)
                                    break
                                end
                            end
                        end

                        if v90 then
                            handle.CFrame = v90.WorldCFrame * attachment.CFrame:Inverse()
                            local weldConstraint = Instance.new("WeldConstraint")
                            weldConstraint.Part0 = handle
                            weldConstraint.Part1 = v90.Parent
                            weldConstraint.Parent = handle
                        else
                            local torso = head or character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso")
                            if torso then
                                handle.CFrame = torso.CFrame
                                local weldConstraint = Instance.new("WeldConstraint")
                                weldConstraint.Part0 = handle
                                weldConstraint.Part1 = torso
                                weldConstraint.Parent = handle
                            end
                        end
                    end
                end
            elseif child:IsA("Clothing") or child:IsA("ShirtGraphic") then
                child:Clone().Parent = character
            end
        end

        result:Destroy()
    end
end

game:GetService("Players").LocalPlayer.CharacterAdded:Connect(function(character)
    if not MorphActiveState or TargetMorphUser == "" then return end
    task.wait(0.5)
    if MorphActiveState and TargetMorphUser ~= "" and game:GetService("Players").LocalPlayer.Character == character then
        pcall(function() ApplyAvatarMorph(TargetMorphUser) end)
    end
end)

--=================================================================
-- ESP ABILITY OVERLAY  (v2.2 — fixed lifecycle)
-- Bugs fixed:
--   1. Stale `char` closure: the update loop now reads plr.Character
--      every tick so a respawned character is always current.
--   2. Billboard floating under map: Adornee is re-pointed to the
--      new Head on every respawn; old BillboardGui is destroyed first.
--   3. Respawn hook missing for players already in-server when ESP
--      was turned on: every player gets a CharacterAdded hook upfront.
--   4. "Can't turn off": connections are stored in a table and
--      disconnected explicitly — no longer relies on task.cancel
--      reaching a Disconnect call it never gets to.
--=================================================================
local ESP_Storage      = {}   -- [plr] = BillboardGui
local ESP_RespawnConns = {}   -- [plr] = CharacterAdded connection
local ESP_Active_Task  = nil
local ESP_Running      = false

local DataAbilities = game:GetService("ReplicatedStorage"):WaitForChild("Misc"):WaitForChild("DataAbilities")

local function RemovePlayerESP(plr)
    if ESP_Storage[plr] then
        pcall(function() ESP_Storage[plr]:Destroy() end)
        ESP_Storage[plr] = nil
    end
end

local function DetachRespawnHook(plr)
    if ESP_RespawnConns[plr] then
        pcall(function() ESP_RespawnConns[plr]:Disconnect() end)
        ESP_RespawnConns[plr] = nil
    end
end

local function AttachPlayerESP(plr)
    if plr == LP then return end
    -- Always remove stale billboard before creating a fresh one
    RemovePlayerESP(plr)

    -- Wait for character with a timeout; bail if ESP was turned off meanwhile
    local char = plr.Character
    if not char then
        local ok, result = pcall(function()
            return plr.CharacterAdded:Wait()
        end)
        if not ok or not ESP_Running then return end
        char = result
    end
    if not ESP_Running then return end

    local head = char:WaitForChild("Head", 5)
    if not head or not ESP_Running then return end

    local sGui = (gethui and gethui()) or LP:WaitForChild("PlayerGui")
    local BGui = Instance.new("BillboardGui")
    BGui.Name          = "Illusion_Plr_ESP"
    BGui.AlwaysOnTop   = true
    BGui.Size          = UDim2.new(0, 140, 0, 40)
    BGui.StudsOffset   = Vector3.new(0, 3.0, 0)
    BGui.Adornee       = head
    BGui.Parent        = sGui
    ESP_Storage[plr]   = BGui

    local Icon = Instance.new("ImageLabel")
    Icon.Size                = UDim2.fromOffset(26, 26)
    Icon.Position            = UDim2.fromOffset(5, 7)
    Icon.BackgroundTransparency = 1
    Icon.Image               = "rbxassetid://3944703587"
    Icon.Parent              = BGui

    local StatusLbl = Instance.new("TextLabel")
    StatusLbl.BackgroundTransparency = 1
    StatusLbl.Size           = UDim2.fromOffset(95, 14)
    StatusLbl.Position       = UDim2.fromOffset(36, 5)
    StatusLbl.Text           = "NONE"
    StatusLbl.TextColor3     = Color3.fromRGB(255, 255, 255)
    StatusLbl.TextSize       = 10
    StatusLbl.Font           = Enum.Font.GothamBold
    StatusLbl.TextXAlignment = Enum.TextXAlignment.Left
    StatusLbl.TextStrokeTransparency = 0
    StatusLbl.Parent         = BGui

    local ReadyLbl = Instance.new("TextLabel")
    ReadyLbl.BackgroundTransparency = 1
    ReadyLbl.Size            = UDim2.fromOffset(95, 14)
    ReadyLbl.Position        = UDim2.fromOffset(36, 18)
    ReadyLbl.Text            = "READY"
    ReadyLbl.TextColor3      = Color3.fromRGB(80, 220, 120)
    ReadyLbl.TextSize        = 10
    ReadyLbl.Font            = Enum.Font.GothamBlack
    ReadyLbl.TextXAlignment  = Enum.TextXAlignment.Left
    ReadyLbl.TextStrokeTransparency = 0
    ReadyLbl.Parent          = BGui

    -- Update loop: reads plr.Character every tick — never holds a stale char reference
    task.spawn(function()
        while BGui and BGui.Parent and ESP_Running do
            pcall(function()
                -- Always use the live character, not the closure-captured one
                local liveChar = plr.Character
                if not liveChar then return end

                -- Re-adorn if the player respawned (new Head instance)
                local liveHead = liveChar:FindFirstChild("Head")
                if liveHead and BGui.Adornee ~= liveHead then
                    BGui.Adornee = liveHead
                end

                local rawAbility = plr:GetAttribute("EquippedAbility") or plr:GetAttribute("Ability") or "None"
                StatusLbl.Text = string.upper(tostring(rawAbility))

                local abilityFolder = DataAbilities:FindFirstChild(tostring(rawAbility))
                if abilityFolder then
                    local assetIcon = abilityFolder:GetAttribute("Icon") or abilityFolder:GetAttribute("Icon1")
                    Icon.Image = (assetIcon and assetIcon ~= "") and assetIcon or "rbxassetid://3944703587"
                else
                    Icon.Image = "rbxassetid://3944703587"
                end

                -- Read cooldown/ability state from live character
                local abActive  = liveChar:GetAttribute("AbilityActive") == true
                local cdActive  = liveChar:GetAttribute("CooldownActive") == true
                local cdExpiry  = plr:GetAttribute("CooldownExpiration")
                local cdExpired = cdExpiry and workspace:GetServerTimeNow() < cdExpiry
                local isReady   = not (abActive or cdActive or cdExpired)

                if isReady then
                    ReadyLbl.Text       = "READY"
                    ReadyLbl.TextColor3 = Color3.fromRGB(80, 220, 120)
                else
                    local cdDiff = cdExpiry and (cdExpiry - workspace:GetServerTimeNow()) or 0
                    ReadyLbl.Text       = cdDiff > 0 and string.format("%.1fS", cdDiff) or "COOLDOWN"
                    ReadyLbl.TextColor3 = Color3.fromRGB(220, 100, 80)
                end
            end)
            task.wait(0.2)
        end
        -- Loop exited: clean up this player's billboard
        pcall(function()
            if BGui and BGui.Parent then BGui:Destroy() end
        end)
        if ESP_Storage[plr] == BGui then ESP_Storage[plr] = nil end
    end)
end

local function HookPlayerRespawn(plr)
    DetachRespawnHook(plr)
    ESP_RespawnConns[plr] = plr.CharacterAdded:Connect(function()
        if not ESP_Running then return end
        -- Small delay so the new character fully loads before we attach
        task.delay(0.3, function()
            if ESP_Running then
                task.spawn(AttachPlayerESP, plr)
            end
        end)
    end)
end

local function SetESPAbility(on)
    -- Always stop cleanly first
    ESP_Running = false

    -- Cancel any pending spawn task
    if ESP_Active_Task then
        pcall(function() task.cancel(ESP_Active_Task) end)
        ESP_Active_Task = nil
    end

    -- Disconnect all respawn hooks
    for plr, _ in pairs(ESP_RespawnConns) do DetachRespawnHook(plr) end
    table.clear(ESP_RespawnConns)

    -- Destroy all billboards
    for plr, _ in pairs(ESP_Storage) do RemovePlayerESP(plr) end
    table.clear(ESP_Storage)

    if not on then return end

    ESP_Running = true

    ESP_Active_Task = task.spawn(function()
        local PlayersService = game:GetService("Players")

        -- Hook and attach ESP for every player already in server
        for _, p in ipairs(PlayersService:GetPlayers()) do
            if p ~= LP then
                HookPlayerRespawn(p)
                task.spawn(AttachPlayerESP, p)
            end
        end

        -- New players joining
        local connAdded = PlayersService.PlayerAdded:Connect(function(p)
            if not ESP_Running or p == LP then return end
            HookPlayerRespawn(p)
            task.spawn(AttachPlayerESP, p)
        end)

        -- Players leaving: destroy billboard + remove respawn hook
        local connRemoving = PlayersService.PlayerRemoving:Connect(function(p)
            DetachRespawnHook(p)
            RemovePlayerESP(p)
        end)

        -- Hold until ESP is turned off
        while ESP_Running and shared._IllusionRunning do
            task.wait(0.5)
        end

        -- Clean shutdown: disconnect session-level connections
        pcall(function() connAdded:Disconnect() end)
        pcall(function() connRemoving:Disconnect() end)

        -- Destroy remaining billboards and hooks (defensive — toggle off may have already cleared them)
        for plr, _ in pairs(ESP_RespawnConns) do DetachRespawnHook(plr) end
        table.clear(ESP_RespawnConns)
        for plr, _ in pairs(ESP_Storage) do RemovePlayerESP(plr) end
        table.clear(ESP_Storage)
    end)
end

                            isEquipped = (getgenv().swordFX == currentName)

        for _, t in pairs(UA.Types) do
            local pg = ip and ip[t]; if pg and pg.BeginBatch then pg.BeginBatch() end
            
            -- คืนค่าฟังก์ชัน MarkDirty ดั้งเดิมของตัวเกมกลับมาเพื่อล้างระบบจำลองสวมรอยของสคริปต์
            if pg and UA.Originals["MarkDirty" .. t] then
                pg.MarkDirty = UA.Originals["MarkDirty" .. t]
                UA.Originals["MarkDirty" .. t] = nil
            end
        end
        for item, ch in pairs(UA.ShopChanges) do
            for _, f in pairs(UA.ShopFields) do
                if ch[f[1]] ~= nil then applyShopState(item, f[2], ch[f[1]]) end
            end
        end
        for _, t in pairs(UA.Types) do
            local pg = ip and ip[t]
            if pg then if pg.MarkDirty then pg.MarkDirty(nil,false,true) end; if pg.EndBatch then pg.EndBatch() end end
        end
        
        -- ล้างค่าตัวแปรสกินส่วนกลางทั้งหมดเพื่อถอดดาบและระเบิดจำลองออกทันทีเมื่อปิดระบบ
        getgenv().swordModel = ""
        getgenv().swordFX = ""
        if getgenv().updateSword then pcall(getgenv().updateSword) end
        
        table.clear(UA.ShopState); table.clear(UA.ShopChanges)
        table.clear(UA.ShopItems.Sword); table.clear(UA.ShopItems.Explosion)
    end

    -- ── Inventory item helpers
    UA.IsUnlockItem = function(item)
        return type(item)=="table" and tostring(item.Id or ""):match("^UnlockAll:") ~= nil
    end
    UA.GetOwnedItem = function(cat, name)
        local ids = UA.InventoryClient:FindItems(cat, name)
        local id  = type(ids)=="table" and ids[1]
        local item = id and UA.InventoryClient:GetItem(cat, id)
        if item and not UA.IsUnlockItem(item) then
            local c = table.clone(item); c.Id = c.Id or id; return c
        end
        for _, v in pairs(UA.InventoryClient:GetInventory()[cat] or {}) do
            if v.Name == name and not UA.IsUnlockItem(v) then return v end
        end
    end
    UA.GetRealEquipped = function(cat)
        local eq = UA.InventoryClient:GetEquipped(cat)
        if not UA.IsUnlockItem(eq) then return eq end
        return UA.GetOwnedItem(cat, eq.Name)
    end
    UA.SetEquipped = function(cat, name)
        if not UA.SavedEquipped[cat] then
            UA.Originals["Equipped"..cat] = UA.GetRealEquipped(cat); UA.SavedEquipped[cat]=true
        end
        
        -- ดึงโครงสร้างข้อมูลไอเท็มจำลองที่ถูกต้องสมบูรณ์ (อิงตามวิธีแมตช์ข้อมูลของ Flyte)
        local targetItem = UA.GetOwnedItem(cat, name)
        if not targetItem then
            -- สร้างตารางตรรกะข้อมูลจำลองเสมือนจริงให้ตรงตามโครงสร้างของตัวเกม เพื่อป้องกันโมดูลอาวุธล่องหน
            targetItem = {
                Name = name,
                Id = "UnlockAll:"..cat..":"..name,
                Type = cat,
                Virtual = true
            }
        end
        
        -- ส่งข้อมูลแพ็กเก็ตสวมใส่เข้าสู่ตัวควบคุมกระเป๋าหลัก (InventoryReplion) เพื่อให้ระบบเซิร์ฟเวอร์จำลองยอมรับสิทธิ์
        UA.InventoryReplion:_set({"Equipped", cat}, targetItem)
        
        -- อัปเดตคุณลักษณะแอตทริบิวต์ส่วนกลางของตัวละครเพื่อกระตุ้นให้เอฟเฟกต์และโมเดลสกินสลับทำงานในสนามแข่งจริง
        if cat == "Sword" then
            LP:SetAttribute("CurrentlyEquippedSword", name)
            local char = UA.GetCharacter()
            if char then char:SetAttribute("CurrentlyEquippedSword", name) end
        elseif cat == "Explosion" then
            LP:SetAttribute("CurrentlyEquippedExplosion", name)
            local char = UA.GetCharacter()
            if char then char:SetAttribute("CurrentlyEquippedExplosion", name) end
        end
        
        if UA.Shop and UA.Shop._selectedItem then UA.Shop:Select(UA.Shop._selectedItem, true) end
    end

    UA.ApplySword = function(name)
        local char = UA.GetCharacter(); if not char or not name then return end
        local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid",2)
        if not hum or hum.Health<=0 or not char:FindFirstChild("HumanoidRootPart") then return end

        end
        UA.PatchSwords()
        if UA.SwordState then UA.SwordState.cancel(char) end
        if UA.Swords:EquipSwordTo(char, name)==false then return end
        UA.Slash = UA.GetSlash(name) or UA.Slash
        if not UA.Accessory then
            local acc = char:FindFirstChild("SwordAccessories"); if acc then acc:Destroy() end
            task.defer(function() local a=char:FindFirstChild("SwordAccessories"); if a then a:Destroy() end end)
        end
        local bc = UA.GetBlade(); if bc then UA.PatchBlade(bc); bc:SetSword(name) end

    end
    UA.WatchSword = function(char)
        if not char then return end
        UA.Add(char.DescendantAdded:Connect(function(d)
            local sw = UA.Sword
            if UA.Status~="active" or not sw then return end
            local m = d:FindFirstAncestorOfClass("Model")
            while m and m.Parent ~= char do m = m:FindFirstAncestorOfClass("Model") end
            if not m or not m:GetAttribute("_equippedSword") or m.Name==sw then return end
            if d:IsA("BasePart") or d:IsA("Decal") then d.Transparency=1
            elseif d:IsA("ParticleEmitter") or d:IsA("Trail") then d.Enabled=false; d:Clear()
            elseif d:IsA("Beam") or d:IsA("Fire") or d:IsA("Smoke") or d:IsA("Sparkles") or d:IsA("Light") then d.Enabled=false end
        end))
        UA.Add(char:GetAttributeChangedSignal("CurrentlyEquippedSword"):Connect(function()
            if UA.Status=="active" and UA.Sword and char:GetAttribute("CurrentlyEquippedSword")~=UA.Sword then
                task.defer(UA.ApplySword, UA.Sword)
            end
        end))
        local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid",2)
        if hum then
            local wasDead = false
            UA.Add(hum.HealthChanged:Connect(function(hp)
                if hp <= 0 then wasDead = true; return end
                if wasDead and UA.Status=="active" and UA.Sword then
                    wasDead = false
                    -- respawn: wait for char to fully load
                    local runToken = UA.Run
                    task.delay(1, function()
                        if UA.Status=="active" and UA.Run==runToken and UA.Sword then
                            UA.ApplySword(UA.Sword)
                            if UA.Explosion then
                                LP:SetAttribute("CurrentlyEquippedExplosion", UA.Explosion)
                                local newChar = LP.Character
                                if newChar then newChar:SetAttribute("CurrentlyEquippedExplosion", UA.Explosion) end
                            end
                        end
                    end)
                end
            end))
        end
    end

    UA.PatchExplosion = function()

        -- Flyte param order: (self, expName, cframe, radius, target, attacker, duration, isLocal)
        -- isLocal (color in flyte) = bool true when the attacker is localPlayer
        UA.VFX.PlayExplosion = function(self, expName, cframe, radius, target, attacker, duration, isLocal)
            -- Override explosion skin when we are the attacker
            if UA.Status=="active" and UA.Explosion and isLocal and attacker==LP.Character then
                expName = UA.Explosion
            end
            -- Finisher: play when target is last alive player and we have finisher equipped
            if UA.Status=="active" and isLocal and attacker==LP.Character and UA.Sword and UA.Finishers[UA.Sword] and UA.FinishersController then
                local hrp   = target and target:FindFirstChild("HumanoidRootPart")
                local alive = workspace:FindFirstChild("Alive")
                local queue = alive and alive:GetChildren() or {}
                local others = 0
                for _, p in pairs(queue) do
                    if p ~= target then others = others + 1 end
                end
                if hrp and others <= 1 then
                    UA.FinishersController:PlayFinisher(UA.Sword, attacker, target, hrp.CFrame, workspace:GetServerTimeNow())
                end
            end
            return UA.Originals.PlayExplosion(self, expName, cframe, radius, target, attacker, duration, isLocal)
        end
    end

    -- ── ParrySuccess patch (slash effect override)
    UA.PatchParrySuccess = function()
        if not (hookfunction and getconnections and getinfo) then return end
        local remotes = RS:FindFirstChild("Remotes")
        local swordFX = getgenv().FlyteSwordEffects or {}; getgenv().FlyteSwordEffects=swordFX
        
        local function patchRemote(key, remote)
            if UA.Originals[key] or not remote then return end
            for _, c in pairs(getconnections(remote)) do
                -- ปรับปรุงตรรกะแบบ Flyte: เช็คผ่านอัปแวลูและโมดูลสคริปต์ เพื่อดักจับฟังก์ชันไม่มีชื่อ (Anonymous) ได้อย่างแม่นยำ
                local fn = c.Function
                if fn and (isourclosure(fn) == false) then
                    local info = getinfo(fn)
                    -- เช็คว่าฟังก์ชันมาจากสคริปต์ควบคุมการพาร์รี่เบื้องหลังของตัวเกมหรือไม่ โดยไม่ต้องพึ่งพาชื่อฟังก์ชันตรงๆ
                    if info and info.source and info.source:find("PRY") or (info.name == "parrySuccessAll" or swordFX[key] == fn) then
                        swordFX[key] = fn
                        UA.Originals[key.."Target"] = fn
                        UA.Originals[key] = hookfunction(fn, function(...)
                            local a = table.pack(...)
                            -- บังคับส่งสัญญาณสวมรอย Slash เอฟเฟกต์จำลองให้แสดงผลในสนามแข่งจริง
                            if UA.Status=="active" and UA.Sword and (tostring(a[4])==LP.Name or (type(a[4])=="table" and UA.Clones[a[4].Character])) then
                                local sl = UA.Slash
                                if sl then 
                                    a[1] = sl        -- ส่งรหัสเอฟเฟกต์วงแหวนฟันจำลอง
                                    a[3] = UA.Sword  -- ส่งรหัสชื่อดาบจำลอง
                                end
                            end
                            return UA.Originals[key](table.unpack(a, 1, a.n))
                        end)
                        return true
                    end
                end
            end
            return false
        end
        
        local r1 = remotes and remotes:FindFirstChild("ParrySuccessAll")
        if r1 then patchRemote("Parry_All_Func", r1.OnClientEvent) end
        
        local r2 = remotes and remotes:FindFirstChild("ParrySuccessClient")
        if r2 then 
            local signal = r2:IsA("BindableEvent") and r2.Event or r2.OnClientEvent
            patchRemote("Parry_Client_Func", signal) 
        end
    end

    UA.LoadEmotes = function()
        local inv = UA.InventoryClient:GetInventory().Emote or {}
        if UA.EmoteCatalogLoaded then return inv end
        table.clear(UA.EmoteItems); table.clear(UA.EmoteItemsList)
        table.clear(UA.EmoteDisplays); table.clear(UA.EmoteItemKeys)
        -- Build owned set — real items only, no fakes
        local owned = {}
        for _, v in pairs(inv) do
            if not UA.IsUnlockItem(v) then owned[v.Name] = true end
        end
        local deadline = os.clock() + 0.001
        for _, anim in pairs(UA.EmotesFolder:GetChildren()) do
            local info = anim:IsA("Animation") and UA.ItemInfo.Emote[anim.Name]
            if info then UA.EmoteDisplays[info.DisplayName or info.Name] = anim.Name end
            -- Only add fake item if: is Animation, not already owned, has valid icon, icon isn't the anim itself
            if anim:IsA("Animation") and not owned[anim.Name] and info
              and type(info.Icon)=="string" and info.Icon~="" and info.Icon~="rbxassetid://0" and info.Icon~=anim.AnimationId then
                local fake = { Name=anim.Name, Id="UnlockAll:Emote:"..anim.Name, CreatedAt=0 }
                UA.EmoteItems[anim.Name]     = fake
                table.insert(UA.EmoteItemsList, fake)
                UA.EmoteItemKeys[UA.InventoryShared:ItemToKey("Emote", fake, {"Id"})] = fake.Id
            end
            -- Yield every 1ms to prevent freeze (flyte pattern)
            if deadline <= os.clock() then
                RunService.Heartbeat:Wait()
                deadline = os.clock() + 0.001
            end
        end
        -- Build EmoteInventory map (Id → item) used by CreateInventory
        UA.EmoteInventory = {}
        for _, fake in ipairs(UA.EmoteItemsList) do
            UA.EmoteInventory[fake.Id] = fake
        end
        UA.EmoteCatalogLoaded = true
        return inv
    end
    UA.PatchInventory = function(inv)
        if UA.Originals.CreateInventory then return end
        UA.Originals.EmoteFindItemsWithKey = UA.InventoryClient.FindItemsWithKey
        UA.Originals.EmoteGetItem         = UA.InventoryClient.GetItem
        UA.InventoryClient.FindItemsWithKey = function(self, cat, key, ...)
            -- Try real inventory first; only return fake if nothing found
            local real = UA.Originals.EmoteFindItemsWithKey(self, cat, key, ...)
            if real and #real > 0 then return real end
            local id = cat=="Emote" and UA.EmoteItemKeys[key]
            if UA.Status=="active" and id then return {id} end
            return real
        end
        UA.InventoryClient.GetItem = function(self, cat, id, ...)
            -- Real item takes priority
            local real = UA.Originals.EmoteGetItem(self, cat, id, ...)
            if real and not UA.IsUnlockItem(real) then return real end
            local n = cat=="Emote" and tostring(id):match("^UnlockAll:Emote:(.+)$")
            if UA.Status=="active" and n and UA.EmoteItems[n] then return UA.EmoteItems[n] end
            return real
        end
        UA.Originals.CreateInventory = UA.InventoryController.CreateInventory
        UA.InventoryController.CreateInventory = function(self, cfg)
            if UA.Status=="active" and cfg.InventoryType=="Emote" then
                local nc=table.clone(cfg); nc.WatchOnChange=false
                nc.FindItemsWithKey=function(cat,key)
                    local id=cat=="Emote" and UA.EmoteItemKeys[key]
                    if id then return UA.EmoteInventory[id] and {id} or {} end
                    return UA.InventoryShared:FindItemsWithKey(UA.EmoteCaller,cat,key)
                end
                local orig_slot=nc.OnSlotCreated
                nc.OnSlotCreated=function(inv2,item,slotObj,trove,...)
                    -- รันฟังก์ชันจำลองเดิมของตัวเกมเพื่อสร้างองค์ประกอบพื้นฐาน
                    if orig_slot then orig_slot(inv2,item,slotObj,trove,...) end
                    
                    -- ค้นหาและดักจับผ่าน UI Instance จริงในเกมเพื่อแก้ปัญหาชื่อแอตทริบิวต์ทับซ้อนและแครชคำสั่ง IsA
                    task.spawn(function()
                        if not item or not item.Name then return end
                        
                        -- ค้นหาปุ่มกดจริง (GuiButton) ที่อยู่ภายใต้ออบเจกต์ Slot ของหน้าต่างไอเท็มเกม
                        local realButton = nil
                        if typeof(slotObj) == "Instance" then
                            realButton = slotObj:IsA("GuiButton") and slotObj or slotObj:FindFirstChildWhichIsA("GuiButton", true)
                        elseif type(slotObj) == "table" and typeof(slotObj.Gui) == "Instance" then
                            realButton = slotObj.Gui:IsA("GuiButton") and slotObj.Gui or slotObj.Gui:FindFirstChildWhichIsA("GuiButton", true)
                        end
                        
                        -- หากเจอทางเชื่อมต่อปุ่มกดจริง ให้ทำการเคลียร์และผูกสคริปต์สวมใส่ไอเท็มแบบแยกอิสระ
                        if realButton then
                            if getconnections then
                                pcall(function()
                                    for _, c in pairs(getconnections(realButton.Activated)) do c:Disconnect() end
                                end)
                            end
                            
                            -- ผูกการทำงานสวมใส่ท่าทางจำลองเข้าสู่ Slot ช่องนั้น ๆ โดยตรง ไม่ให้ทับถมกันมั่วซั่ว
                            realButton.Activated:Connect(function()
                                local targetPlayerName = item.Name
                                task.defer(UA.ApplyEmote, UA.GetEmoteSlot(), targetPlayerName)
                            end)
                        end
                    end)
                end
                UA.EmoteView = UA.Originals.CreateInventory(self,nc); return UA.EmoteView
            end
            return UA.Originals.CreateInventory(self,cfg)
        end
        if not UA.EmoteView then
            local orig_sort=UA.InventoryController.CreateSortOptions
            UA.InventoryController.CreateSortOptions=function()end
            
            if UA.EmoteWheel and type(UA.EmoteWheel.populateEmoteList) == "function" then
                UA.EmoteWheel:populateEmoteList()
            elseif UA.EmoteWheel and type(UA.EmoteWheel.populateEmotesList) == "function" then
                UA.EmoteWheel:populateEmotesList()
            else
                pcall(function()
                    UA.InventoryController:CreateSortOptions()
                    UA.EmoteWheel:populateEmoteList()
                end)
            end
            
            UA.InventoryController.CreateSortOptions=orig_sort
        end
        UA.HookEmotes()
    end
    UA.RestoreInventory = function(full)
        UA.EmoteBuildToken=UA.EmoteBuildToken+1; UA.EmoteBuilding=false
        UA.EmoteBuildIndex=1; UA.EmotesReady=false
        onInputEnded("CreateInventory",  UA.InventoryController)
        onInputEnded("EmoteFindItemsWithKey", UA.InventoryClient, "FindItemsWithKey")
        onInputEnded("EmoteGetItem",     UA.InventoryClient, "GetItem")
        if UA.EmoteView then UA.EmoteView.Destroy(); UA.EmoteView=nil end
        if UA.EmoteCaller then UA.EmoteCaller.Replion:Destroy(); UA.EmoteCaller=nil end
        if UA.EmoteContainer then UA.EmoteContainer:Destroy(); UA.EmoteContainer=nil end
        UA.EmoteSource=nil; UA.EmoteInventory=nil; UA.EmoteCatalogLoaded=false
    end
    UA.HookEmotes = function()
        local scroller = LP.PlayerGui:FindFirstChild("EmoteWheel")
        scroller = scroller and scroller:FindFirstChild("List") and scroller.List:FindFirstChild("Content")
        if not scroller then return end
        local function hookCard(c)
            if UA.Status~="active" or not c:IsA("GuiButton") or not c:FindFirstChild("ItemName") or UA.CardConnections[c] then return end
            UA.CardConnections[c]=c.ItemName:GetPropertyChangedSignal("Text"):Connect(function()
                local name=c.ItemName.Text; local id=UA.EmoteDisplays[name]
                if id and UA.EmoteItems[id] then
                    if getconnections then
                        local fav=c:FindFirstChild("Favorite"); local del=c:FindFirstChild("Delete")
                        if fav then for _,cn in getconnections(fav.Activated) do cn:Disconnect() end end
                        if del then for _,cn in getconnections(del.Activated) do cn:Disconnect() end end
                    end
                    local lines={}
                    local fav=c:FindFirstChild("Favorite"); local del=c:FindFirstChild("Delete")
                    if fav then table.insert(lines,fav.Activated:Connect(function() UA.ToggleFavorite("Emote",id) end)) end
                    if del then table.insert(lines,del.Activated:Connect(function() -- delete emote slot
                        UA.Deleted.Emote[id]=true; UA.EquippedEmotes[UA.GetEmoteSlot()]=nil; UA.UpdateEmoteWheel(); UA._saveCfg()
                    end)) end
                    UA.ButtonConnections[c]=lines
                end
            end)
        end
        for _, d in scroller:GetDescendants() do hookCard(d) end
        UA.Add(scroller.DescendantAdded:Connect(function(d) task.defer(hookCard,d) end))
    end
    UA.RestoreEmotes = function()
        table.clear(UA.EquippedEmotes); UA.StopEmote()
        UA.EmoteWheel:UpdateHolderEmotes(); UA.EmoteWheel:UpdateMenuContentEmotes()
    end

    -- ── Defaults / Apply saved
    UA.SaveDefaults = function()
        local m=LP.Character
        UA.Defaults.Sword = (m and m:GetAttribute("CurrentlyEquippedSword")) or LP:GetAttribute("CurrentlyEquippedSword") or "Base Sword"
        UA.Defaults.Explosion = LP:GetAttribute("CurrentlyEquippedExplosion") or (m and m:GetAttribute("CurrentlyEquippedExplosion")) or "Explosion Normal"
        UA.Defaults.Accessory = LP:GetAttribute("ShowSwordAccessory")==true
    end
    UA.ApplySaved = function()
        local sw = tostring(UA._cfg.Sword or "")
        sw = UA.AwakenBase[sw] or sw
        local ex = tostring(UA._cfg.Explosion or "")
        UA.Accessory = UA._cfg.Accessory==true; LP:SetAttribute("ShowSwordAccessory", UA.Accessory)
        if sw~="" and (not UA.ShopActive or UA.ShopItems.Sword[sw]) then UA.SelectSword(sw,false) end
        if ex~="" and (not UA.ShopActive or UA.ShopItems.Explosion[ex]) then UA.SelectExplosion(ex,false) end
        table.clear(UA.EquippedEmotes)
        local owned={}; for _,v in pairs(UA.InventoryClient:GetInventory().Emote or {}) do owned[v.Name]=true end
        local emotes=UA._cfg.Emotes or {}; local slots={}
        for k in pairs(emotes) do local n=tonumber(k); if n then table.insert(slots,n) end end
        table.sort(slots)
        for _, s in pairs(slots) do
            local name = emotes[tostring(s)] or emotes[s]
            if name and (owned[name] or UA.EmoteItems[name]) then UA.EquippedEmotes[s]=name end
        end
        table.clear(UA.Finishers); table.clear(UA.DefaultFinishers)
        for k,v in pairs(UA._cfg.Finishers or {}) do
            if v==true then
                UA.Finishers[k]=true
                if UA.DataReplion then
                    local path={"Finishers","Equipped",k}
                    UA.DefaultFinishers[k]=UA.DataReplion:Get(path)==true
                    UA.DataReplion:_set(path,true)
                end
            end
        end
        UA.UpdateEmoteWheel()
    end
    UA._saveCfg = function()
        UA._cfg.Sword = UA.AwakenBase[UA.Sword] or UA.Sword or ""
        UA._cfg.Explosion = UA.Explosion or ""
        UA._cfg.Accessory = UA.Accessory==true
        local em={}; for s,n in pairs(UA.EquippedEmotes) do em[tostring(s)]=n end
        UA._cfg.Emotes = em
        UA._cfg.Finishers = table.clone(UA.Finishers)
        UA._cfg.Favorites = {
            Sword=table.clone(UA.Favorites.Sword),
            Explosion=table.clone(UA.Favorites.Explosion),
            Emote=table.clone(UA.Favorites.Emote),
        }
    end

    -- ── Stop / Start / Toggle
    UA.Stop = function(restoreAll)
        if restoreAll then
            for _, k in pairs({"Parry_All_Func","Parry_Client_Func"}) do
                local tgt = UA.Originals[k.."Target"]
                if tgt then hookfunction(tgt, UA.Originals[k]); UA.Originals[k.."Target"]=nil; UA.Originals[k]=nil end
            end
        end
        local prev=UA.Status; UA.Status=nil; UA.Run=UA.Run+1
        if prev~="active" then
            if restoreAll then UA.RestoreInventory(true) end; return
        end
        onInputEnded("UpdateMultiDelete", UA.Shop, "_updateMultiDelete")
        onInputEnded("CanDelete", UA.Shop, "_canDelete")
        for _, t in pairs(UA.Types) do
            local pg=UA.Shop._inventoryPages and UA.Shop._inventoryPages[t]
            if pg then onInputEnded("MarkDirty"..t, pg, "MarkDirty") end
        end
        cleanConn(UA.Connections); table.clear(UA.Connections)
        cleanConn(UA.CardConnections); table.clear(UA.CardConnections)
        UA.RestoreInventory(restoreAll)
        UA.RestoreShop()
        UA.RestoreFavorites()
        LP:SetAttribute("ShowSwordAccessory", UA.Defaults.Accessory==true)
        for k, def in pairs(UA.DefaultFinishers) do
            if UA.DataReplion then UA.DataReplion:_set({"Finishers","Equipped",k},def==true) end
        end
        table.clear(UA.Finishers); table.clear(UA.DefaultFinishers)
        onInputEnded("PlayExplosion", UA.VFX)
        onInputEnded("EmotePlay", UA.EmoteController, "Play")
        onInputEnded("UpdateHolderEmotes", UA.EmoteWheel)
        onInputEnded("UpdateMenuContentEmotes", UA.EmoteWheel)
        UA.RestoreSword(); UA.RestoreExplosion()
        UA.RestoreEmotes()
        if UA.Originals.SetSword and UA.BladeController then onInputEnded("SetSword", UA.BladeController) end
    end

    UA.Start = function()
        if UA.Status then return end
        UA.Status="starting"; local RUN=UA.Run+1; UA.Run=RUN
        if not UA.Load() then UA.Status=nil; return end
        while UA._enabled and UA.Run==RUN and not UA.ShopReady() do RunService.Heartbeat:Wait() end
        if not UA._enabled or UA.Run~=RUN then if UA.Run==RUN then UA.Status=nil end; return end
        UA.Status="active"
        for _, d in pairs(UA.Deleted) do table.clear(d) end
        table.clear(UA.EquippedEmotes); table.clear(UA.SavedEquipped)
        table.clear(UA.Finishers); table.clear(UA.DefaultFinishers)
        UA.Accessory=false; UA.Originals.EquippedSword=nil; UA.Originals.EquippedExplosion=nil
        UA.LoadFavorites(); UA.SaveDefaults()
        local inv=UA.LoadEmotes()
        if UA.Run~=RUN or UA.Status~="active" then return end
        UA.PatchInventory(inv); UA.HookEmotes()
        
        -- =================================================================
        -- 🛠️ แกนบังคับสวมรอยแพ็กเก็ตและวาดปุ่มเสมือนจริงในสนามแข่ง (FIXED 2026)
        -- =================================================================
        task.spawn(function()
            local function GetShopGui()
                local pGui = LP:FindFirstChild("PlayerGui")
                if not pGui then return nil end
                return pGui:FindFirstChild("Shop") or pGui:FindFirstChild("ControllerShop")
            end

            -- ฟังก์ชันบังคับวาดปุ่ม EQUIP และเปลี่ยนแอตทริบิวต์จริงระดับเซิร์ฟเวอร์จำลอง
            local function ForceRenderEquipButton()
                pcall(function()
                    local currentGui = GetShopGui()
                    local infoBg = currentGui and currentGui:FindFirstChild("Holder") and currentGui.Holder:FindFirstChild("InfoBG")
                    local buyButton = infoBg and infoBg:FindFirstChild("BuyButton")
                    local priceTag = buyButton and buyButton:FindFirstChild("PriceTag")
                    local btnText = priceTag and priceTag:FindFirstChild("TextLabel")
                    local priceLabel = priceTag and priceTag:FindFirstChild("Price")

                    if UA.Shop and UA.Shop._selectedItem and buyButton and btnText then
                        local selected = UA.Shop._selectedItem
                        local cat = selected.type   -- "Sword" หรือ "Explosion"
                        local name = selected.name  -- ชื่อไอเท็ม
                        
                        local itemObj = UA.ShopItems[cat] and UA.ShopItems[cat][name]
                        local state = itemObj and UA.ShopState[itemObj]

                        if state and state.Virtual then
                            buyButton.Visible = true
                            buyButton.Active = true
                            if priceLabel then priceLabel.Visible = false end
                            
                            -- เช็คสถานะการใส่ดาบผ่านแอตทริบิวต์หลักของ LocalPlayer และ Global Variable พร้อมกัน
                            local isEquipped = false
                            if cat == "Sword" then
                                isEquipped = (getgenv().swordModel == name) or (LP:GetAttribute("CurrentlyEquippedSword") == name)
                            elseif cat == "Explosion" then
                                isEquipped = (getgenv().swordFX == name) or (LP:GetAttribute("CurrentlyEquippedExplosion") == name)
                            end
                            
                            if isEquipped then
                                btnText.Text = "Equipped"
                                buyButton.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
                            else
                                btnText.Text = "Equip"
                                buyButton.BackgroundColor3 = Color3.fromRGB(125, 90, 220)
                            end
                        end
                    end
                end)
            end

            -- HOOK คุมคำสั่งกดปุ่มเลือกไอเท็มในร้าน
            if UA.Shop and UA.Shop.Select then
                local orig_Select = UA.Shop.Select
                UA.Shop.Select = function(self, itemData, ...)
                    local res = orig_Select(self, itemData, ...)
                    if UA.Status == "active" then
                        task.defer(ForceRenderEquipButton)
                    end
                    return res
                end
            end

            -- ลูปดักจับคำสั่งกดสวมใส่เสมือนจริง ยิงข้ามไปยังระบบควบคุมดาบของตัวเกมหลัก
            task.spawn(function()
                while UA.Status == "active" or UA.Status == "starting" do
                    task.wait(0.2)
                    local currentGui = GetShopGui()
                    local infoBg = currentGui and currentGui:FindFirstChild("Holder") and currentGui.Holder:FindFirstChild("InfoBG")
                    local buyButton = infoBg and infoBg:FindFirstChild("BuyButton")
                    
                    if buyButton and not buyButton:GetAttribute("IllusionHooked") then
                        buyButton:SetAttribute("IllusionHooked", true)
                        UA.Add(buyButton.Activated:Connect(function()
                            if UA.Shop and UA.Shop._selectedItem then
                                local selected = UA.Shop._selectedItem
                                local cat = selected.type
                                local name = selected.name
                                local itemObj = UA.ShopItems[cat] and UA.ShopItems[cat][name]
                                
                                if itemObj and UA.ShopState[itemObj] and UA.ShopState[itemObj].Virtual then
                                    if cat == "Sword" then
                                        -- บังคับล้างดาบเก่าทั้งหมดในสนามแข่งขันเพื่อป้องกันการวาดทับซ้อน
                                        pcall(function()
                                            if LP.Character then
                                                for _, v in ipairs(LP.Character:GetChildren()) do
                                                    if v:IsA("Model") and RS.Shared.ReplicatedInstances.Swords:GetSword(v.Name) then
                                                        v:Destroy()
                                                    end
                                                end
                                            end
                                        end)

                                        getgenv().swordModel = name
                                        LP:SetAttribute("CurrentlyEquippedSword", name)
                                        if LP.Character then LP.Character:SetAttribute("CurrentlyEquippedSword", name) end
                                        if getgenv().updateSword then pcall(getgenv().updateSword) end
                                        pcall(function() UA.ApplySword(name) end)
                                    elseif cat == "Explosion" then
                                        getgenv().swordFX = name
                                        LP:SetAttribute("CurrentlyEquippedExplosion", name)
                                        if LP.Character then LP.Character:SetAttribute("CurrentlyEquippedExplosion", name) end
                                    end
                                    
                                    task.defer(ForceRenderEquipButton)
                                end
                            end
                        end))
                    end
                end
            end)
        end)

        -- รันระบบเอฟเฟกต์จำลองและสกินต่อตามปกติ
        UA.PatchExplosion(); UA.PatchEmotes(); UA.PatchParrySuccess()
        UA.ActivateShop(); UA.ApplySaved()
        UA.WatchSword(LP.Character)
        local CS=game:GetService("CollectionService")
        UA.Add(CS:GetInstanceAddedSignal("EncryptedClone"):Connect(function(m) end))
        if UA.WatchEmotes then UA.WatchEmotes(LP.Character) end
        UA.Add(LP:GetAttributeChangedSignal("CurrentlyEquippedSword"):Connect(function()
            if UA.Sword and LP:GetAttribute("CurrentlyEquippedSword")~=UA.Sword then task.defer(UA.ApplySword,UA.Sword) end
        end))
        UA.Add(LP.CharacterAdded:Connect(function(char)
            local runToken = UA.Run
            UA.WatchSword(char)
            if UA.WatchEmotes then UA.WatchEmotes(char) end
            -- Wait 1s for character to fully load before applying (flyte timing)
            task.delay(1, function()
                if UA.Status~="active" or UA.Run~=runToken or LP.Character~=char then return end
                if UA.Sword then UA.ApplySword(UA.Sword) end
                if UA.Explosion then
                    LP:SetAttribute("CurrentlyEquippedExplosion", UA.Explosion)
                    char:SetAttribute("CurrentlyEquippedExplosion", UA.Explosion)
                end
            end)
        end))
    end

    UA.Toggle = function(on)
        UA._enabled=on
        if on then task.defer(UA.Start) else UA.Stop(true) end
    end

    -- ── Accessory / Finisher helpers
    UA.SetAccessory = function(on)
        UA.Accessory=on==true; LP:SetAttribute("ShowSwordAccessory",UA.Accessory)
        if UA.Accessory and UA.Sword then UA.ApplySword(UA.Sword)

        else
            UA.Finishers[name]=true
            if UA.DataReplion then
                UA.DefaultFinishers[name]=UA.DataReplion:Get({"Finishers","Equipped",name})==true
                UA.DataReplion:_set({"Finishers","Equipped",name},true)
            end
        end
        UA._saveCfg()
    end
    UA.WatchEmotes = function(char)
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid",2)
        if not hum then return end
        UA.Add(hum:GetPropertyChangedSignal("MoveDirection"):Connect(function()
            if hum.MoveDirection~=Vector3.zero and UA.EmoteTrove then UA.StopEmote() end
        end))
    end

-- Float drag + click
function MakeDrag(btn, onTap)
    local d = false
    local ds = nil
    local sp = nil
    local moved = false
    
    btn.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            d = true
            moved = false
            ds = i.Position
            sp = btn.Position
        end
    end)
    
    UIS.InputChanged:Connect(function(i)
        if not d then return end
        if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
            local dl = i.Position - ds
            if dl.Magnitude > 6 then 
                moved = true 
            end
            if moved then
                local vp = workspace.CurrentCamera.ViewportSize
                btn.Position = UDim2.new(
                    sp.X.Scale,
                    math.clamp(sp.X.Offset + dl.X, 0, vp.X - btn.AbsoluteSize.X),
                    sp.Y.Scale,
                    math.clamp(sp.Y.Offset + dl.Y, 0, vp.Y - btn.AbsoluteSize.Y)
                )
            end
        end
    end)
    
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            if d then
                d = false
                if not moved then 
                    xpcall(function()
                        onTap()
                    end, function(err)
                        print("Float button tap action failed: " .. tostring(err))
                    end)
                end
            end
        end
    end)
end

    local s = shared._SetSpamActive

end)

local Opened = false
local function baseScale()
	local vp = Gui.AbsoluteSize
	if vp.X == 0 then return 1 end
	return math.clamp(math.min(vp.X / 620, vp.Y / 400), .45, 1)
end
local function setOpen(v)

	if v then
		Main.Visible = true

		tw.Completed:Connect(function() 
			if not Opened then

			end 
		end)
	end
end

-- อัพเดททุกเฟรม
local startTime, frames, acc = os.time(), 0, 0
local function fmtTime(s)
	return string.format("%02d:%02d:%02d", math.floor(s / 3600), math.floor(s % 3600 / 60), s % 60)
end
connect(RunService.RenderStepped, function(dt)
	if State.Animated then
		StrokeGrad.Rotation = (StrokeGrad.Rotation + dt * 70) % 360
		BgGrad.Rotation = 45 + math.sin(os.clock() * .6) * 30
		for _, s in ipairs(spinList) do
			s[1].Rotation = (s[1].Rotation + dt * s[2]) % 360
		end
	end
	frames += 1
	acc += dt
	if acc >= .5 then
		cards.FPS.Text = tostring(math.floor(frames / acc + .5))
		local ok, ping = pcall(function() return math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) end)
		cards.Ping.Text = ok and (ping .. " ms") or "N/A"
		cards.Players.Text = #Players:GetPlayers() .. " / " .. Players.MaxPlayers
		cards.Session.Text = fmtTime(os.time() - startTime)
		frames, acc = 0, 0
	end
end)

-- ปิดสคริปต์
unload = function()
	-- 1. บันทึกค่าคอนฟิกล่าสุดลง JSON ก่อนเคลียร์ค่าปิดสคริปต์ (ระบบ Auto Save ก่อนปิด)
	if SaveConfig then pcall(SaveConfig) end

	-- 2. รัน For Loop เคลียร์ค่าในตาราง Cfg ทั้งหมดที่เป็น Boolean ให้กลับเป็น false
	if type(Cfg) == "table" then
		for key, value in pairs(Cfg) do
			if type(value) == "boolean" then
				Cfg[key] = false
			end
		end
	end

	-- 3. สั่งปิดฟังก์ชันคำนวณและ Loop เบื้องหลังของระบบต่างๆ ในเกมทันที
	shared._IllusionRunning = false
	if shared._IllusionUnlockAll then
		pcall(function() shared._IllusionUnlockAll.Toggle(false) end) -- ปิดระบบ Unlock All สกินทั้งหมด
	end
	if type(StartParryLoop) == "function" then pcall(StartParryLoop) end -- อัปเดตหยุดทำงาน Parry Loop
	if type(SetTriggerbot) == "function" then pcall(SetTriggerbot, false) end -- ปิด Triggerbot
	if type(SetBallStats) == "function" then pcall(SetBallStats, false) end -- ปิดหน้าต่าง Ball Stats
	if type(SetESPAbility) == "function" then pcall(SetESPAbility, false) end -- ปิดระบบ ESP
	if type(SetNoRender) == "function" then pcall(SetNoRender, false) end -- คืนค่า Render ตัวเกม
	if type(StopFly) == "function" then pcall(StopFly) end -- คืนค่าระบบบินสระตัวละคร
	if type(SetOrbit) == "function" then pcall(SetOrbit, false) end -- ปิดระบบ Orbit หมุน

	-- 4. ล้างคุณลักษณะทางกายภาพเดิมของแม่แบบ GUI หลัก
	setFly(false)
	setNoclip(false)
	setFullbright(false)
	setNoFog(false)
	
	-- 5. ทำลายการเชื่อมต่อ (Connections) และลบหน้าต่าง UI ทิ้งอย่างสมบูรณ์
	for _, c in ipairs(Connections) do pcall(function() c:Disconnect() end) end
	if Opened then
		Opened = false
		tween(Scale, .25, {Scale = 0}, Enum.EasingStyle.Back, Enum.EasingDirection.In)
		task.wait(.3)
	end
	Gui:Destroy()
	_G.INVIS_HUB_UNLOAD = nil
end
