local cleanupResources
cleanupResources = function(thread, seen)
    if typeof(thread) == "RBXScriptConnection" then
        pcall(function() thread:Disconnect() end)
    elseif type(thread) == "thread" and coroutine.status(thread) ~= "dead" then
        pcall(task.cancel, thread)
    elseif type(thread) == "table" then
        seen = seen or {}
        if seen[thread] then return end
        seen[thread] = true
        for _, v in pairs(thread) do cleanupResources(v, seen) end
    end
end

local function callWithRestoredThreadIdentity(fn, ...)
    if type(fn) ~= "function" then return end
    local id = getthreadidentity and getthreadidentity() or nil
    local args = table.pack(...)
    pcall(fn, table.unpack(args, 1, args.n))
    if id and setthreadidentity then setthreadidentity(id) end
end

					hook.DataReplion:_set({ "Finishers", "Equipped", vertex }, particle == true)
				end
			end

			table.clear(hook.Finishers)
			table.clear(hook.DefaultFinishers)
			onInputEnded("PlayExplosion", hook.VFX)
			onInputEnded("EmotePlay", hook.EmoteController, "Play")
			onInputEnded("UpdateHolderEmotes", hook.EmoteWheel)
			onInputEnded("UpdateMenuContentEmotes", hook.EmoteWheel)
			hook.RestoreSword()

			if hook.Originals.SetSword and hook.BladeController then
				onInputEnded("SetSword", hook.BladeController)
			end

			hook.RestoreExplosion()
			hook.RestoreEmotes()
			hook.Sword = nil
			hook.Explosion = nil
		end

		hook.Toggle = function(player28)
			flags.UnlockAllEnabled = player28

			if player28 then
				task.defer(hook.Start)
			else
				hook.Stop()
			end
		end

-- Auto-fill emotes if Cfg.emotes == 'all'

-- ================================================================
-- Auto-fill emote slots when Cfg.emotes = "all"
-- Called after LoadEmotes builds EmoteItemsList
-- ================================================================
local hook = _G._UnlockAllHook
local _originalLoadEmotes = hook.LoadEmotes
hook.LoadEmotes = function()
    local result = _originalLoadEmotes()
    if Cfg.emotes == "all" then
        local owned = {}
        local inv = {}
        pcall(function()
            inv = hook.InventoryClient:GetInventory().Emote or {}
            for _, item in pairs(inv) do
                if not hook.IsUnlockItem(item) then owned[item.Name] = true end
            end
        end)
        local MAX, slot = 24, 1
        table.clear(flags.UnlockAllEmotes)
        for _, item in pairs(inv) do
            if slot > MAX then break end
            if not hook.IsUnlockItem(item) then
                flags.UnlockAllEmotes[tostring(slot)] = item.Name
                slot = slot + 1
            end
        end
        for _, entry in pairs(hook.EmoteItemsList or {}) do
            if slot > MAX then break end
            flags.UnlockAllEmotes[tostring(slot)] = entry.Name
            slot = slot + 1
        end
        print(string.format("[UnlockAll] Auto-filled %d emote slots", slot - 1))
        configData.UnlockAll.Emotes = flags.UnlockAllEmotes
    end
    return result
end

-- ================================================================
-- Auto-detect sword/explosion from game if Cfg is empty
-- ================================================================
if Cfg.sword == "" then
    local function getSword()
        local char = localPlayer.Character
        return (char and char:GetAttribute("CurrentlyEquippedSword"))
            or localPlayer:GetAttribute("CurrentlyEquippedSword") or ""
    end
    local detected = getSword()
    if detected ~= "" then
        flags.UnlockAllSword = detected
        configData.UnlockAll.Sword = detected
    end
    localPlayer:GetAttributeChangedSignal("CurrentlyEquippedSword"):Connect(function()
        if hook.Status ~= "active" then return end
        local n = localPlayer:GetAttribute("CurrentlyEquippedSword") or ""
        if n ~= "" and n ~= hook.Sword then
            flags.UnlockAllSword = n
            configData.UnlockAll.Sword = n
        end
    end)
end
