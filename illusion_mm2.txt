local function W2V(pos)
    local v, on = Camera:WorldToViewportPoint(pos)
    return Vector2.new(v.X, v.Y), on, v.Z
end

local function MakeDrag(frame, handle)
    local d, ds, sp = false
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            d = true; ds = i.Position; sp = frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if d and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local dl = i.Position - ds
            frame.Position = UDim2.new(sp.X.Scale, sp.X.Offset + dl.X, sp.Y.Scale, sp.Y.Offset + dl.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then d = false end
    end)
end

    VL.Position = UDim2.new(1, -50, 0, 5); VL.Size = UDim2.fromOffset(40, 16)

    TBG.Size = UDim2.new(1, -20, 0, 4); TBG.Position = UDim2.fromOffset(10, 32)
    TBG.BackgroundColor3 = C.Card2; TBG.Parent = Row; Rnd(TBG, 4)

    local Fill = Instance.new("Frame")
    Fill.BackgroundColor3 = C.Purple
    Fill.Size = UDim2.new((val - min) / (max - min), 0, 1, 0); Fill.Parent = TBG; Rnd(Fill, 4)

    local drg = false
    local function Update(px)
        local pct = math.clamp((px - TBG.AbsolutePosition.X) / TBG.AbsoluteSize.X, 0, 1)

        Fill.Size = UDim2.new(pct, 0, 1, 0)
        VL.Text = (fmt or "%d"):format(val)
        if onChange then onChange(val) end
    end

    UserInputService.InputChanged:Connect(function(i) if drg and i.UserInputType == Enum.UserInputType.MouseMovement then Update(i.Position.X) end end)
    UserInputService.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 then drg = false end end)

local SilentTarget = nil
local function GetRole(player)
    if not player or not player.Parent then return "innocent" end
    local char = player.Character
    local bag  = player:FindFirstChildOfClass("Backpack")

    local toolList = {}
    if char then for _, t in pairs(char:GetChildren()) do if t:IsA("Tool") then table.insert(toolList, t) end end end
    if bag then for _, t in pairs(bag:GetChildren()) do if t:IsA("Tool") then table.insert(toolList, t) end end end

    for _, tool in pairs(toolList) do
        local n = tool.Name:lower()
        if n:find("knife") or n == "ghostify" or n:find("blade") then return "murderer" end
        if n == "sheriff" or n:find("revolver") or n:find("gun") or n:find("pistol") then return "sheriff" end
    end
    return "innocent"
end

local BodyVelocityInstance, BodyGyroInstance = nil, nil
function ToggleFly(state)
    if state then
        local char = LP.Character or LP.CharacterAdded:Wait()
        local root = char:WaitForChild("HumanoidRootPart", 5)
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not root or not hum then return end

        BodyVelocityInstance = Instance.new("BodyVelocity")
        BodyVelocityInstance.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        BodyVelocityInstance.Velocity = Vector3.new(0, 0, 0)
        BodyVelocityInstance.Parent = root

        BodyGyroInstance = Instance.new("BodyGyro")
        BodyGyroInstance.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        BodyGyroInstance.P = 9e4
        BodyGyroInstance.Parent = root

        task.spawn(function()
            while Cfg.FlyOn and root and hum and BodyVelocityInstance and BodyGyroInstance do
                hum.PlatformStand = true
                BodyGyroInstance.CFrame = Camera.CFrame
                local moveVector = hum.MoveDirection
                BodyVelocityInstance.Velocity = Camera.CFrame:VectorToWorldSpace(moveVector) * Cfg.FlySpeed
                RunService.RenderStepped:Wait()
            end
            if BodyVelocityInstance then BodyVelocityInstance:Destroy() end
            if BodyGyroInstance then BodyGyroInstance:Destroy() end
            if hum then hum.PlatformStand = false end
        end)
    else
        if BodyVelocityInstance then BodyVelocityInstance:Destroy() end
        if BodyGyroInstance then BodyGyroInstance:Destroy() end
        if LP.Character and LP.Character:FindFirstChildOfClass("Humanoid") then
            LP.Character:FindFirstChildOfClass("Humanoid").PlatformStand = false
        end
    end
end

--// ============================================================

local isTweeningCoin = false
task.spawn(function()
    while Gui and Gui.Parent do
        task.wait(0.1)
        if Cfg.AutoCoinOn and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = LP.Character.HumanoidRootPart
            local hum = LP.Character:FindFirstChildOfClass("Humanoid")

            if hum and hum.Health > 0 and not isTweeningCoin then
                local targetCoin = nil
                local minDist = math.huge

                for _, obj in pairs(workspace:GetDescendants()) do
                    if obj.Name == "Coin_Server" or obj.Name:lower():find("coin") then
                        if obj:IsA("BasePart") or (obj:IsA("Model") and obj.PrimaryPart) then
                            local pos = obj:IsA("BasePart") and obj.Position or obj.PrimaryPart.Position
                            local dist = (hrp.Position - pos).Magnitude
                            if dist < minDist then
                                minDist = dist
                                targetCoin = obj
                            end
                        end
                    end
                end

                if targetCoin and targetCoin.Parent then
                    isTweeningCoin = true
                    local originalNoclip = Cfg.NoclipOn
                    Cfg.NoclipOn = true

                    local cPos = targetCoin:IsA("BasePart") and targetCoin.Position or targetCoin.PrimaryPart.Position
                    local dist = (hrp.Position - cPos).Magnitude
                    local duration = math.clamp(dist / Cfg.CoinSpeed, 0.05, 4)

                    local coinTween = Tw(hrp, duration, {CFrame = CFrame.new(cPos)})
                    if coinTween then
                        coinTween:Play()

                        -- Check every frame if coin touched or deleted
                        while coinTween.PlaybackState == Enum.PlaybackState.Playing and targetCoin and targetCoin.Parent do
                            task.wait()
                            local curPos = targetCoin:IsA("BasePart") and targetCoin.Position or targetCoin.PrimaryPart.Position
                            if (hrp.Position - curPos).Magnitude <= 2.5 then
                                coinTween:Cancel() -- Cancel tween immediately on hit!
                                break
                            end
                        end
                    end

                    if not originalNoclip then Cfg.NoclipOn = false end
                    isTweeningCoin = false
                end
            end
        end
    end
end)

--// KEEP GUN (Teleport to Dropped Gun & Teleport Back 0.2s)
--// ============================================================

local isGrabbingGun = false
task.spawn(function()
    while Gui and Gui.Parent do
        task.wait(0.2)
        if Cfg.KeepGunOn and not isGrabbingGun and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = LP.Character.HumanoidRootPart
            local hum = LP.Character:FindFirstChildOfClass("Humanoid")

            if hum and hum.Health > 0 then
                local foundGun = nil

                for _, mapFolder in pairs(workspace:GetChildren()) do
                    local gunDrop = mapFolder:FindFirstChild("GunDrop")
                    if gunDrop and gunDrop:IsA("BasePart") then
                        foundGun = gunDrop
                        break
                    end
                end

                if foundGun then
                    isGrabbingGun = true
                    local oldPos = hrp.CFrame

                    -- Instant Teleport to Gun Drop
                    hrp.CFrame = foundGun.CFrame + Vector3.new(0, 1, 0)
                    task.wait(0.2)

                    -- Teleport back to original position
                    if hrp and hum and hum.Health > 0 then
                        hrp.CFrame = oldPos
                    end
                    task.wait(0.5)
                    isGrabbingGun = false
                end
            end
        end
    end
end)

function ClearRoleESP()
    for _, t in pairs(ESPObjects) do for _, d in pairs(t) do pcall(function() d:Remove() end) end end
    ESPObjects = {}
end

function ClearGunESP()
    for _, d in pairs(GunESPObjects) do pcall(function() d:Remove() end) end
    GunESPObjects = {}
end

Players.PlayerRemoving:Connect(function(player)
    if player == LP then getgenv().ILLUSION_HUB_LOADED = nil end
end)

Gui.AncestryChanged:Connect(function()
    if not Gui.Parent then
        getgenv().ILLUSION_HUB_LOADED = nil
        ClearRoleESP()
        ClearGunESP()
        if ShootBtn then ShootBtn:Destroy() end
    end
end)
