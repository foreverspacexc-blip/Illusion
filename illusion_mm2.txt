local function Tw(obj, t, props)
    TweenService:Create(obj, TweenInfo.new(t, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), props):Play()
end

local function Rnd(obj, r)
    local u = Instance.new("UICorner"); u.CornerRadius = UDim.new(0, r or 8); u.Parent = obj
end

local function Str(obj, col, th)
    local s = Instance.new("UIStroke")
    s.Color = col or C.Stroke; s.Thickness = th or 1
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = obj
end

local function Lbl(par, txt, sz, col, ax)
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Text = txt or ""; l.TextSize = sz or 13
    l.TextColor3 = col or C.Text
    l.Font = Enum.Font.GothamMedium

end

local function W2V(pos)
    local v, on = Camera:WorldToViewportPoint(pos)
    return Vector2.new(v.X, v.Y), on, v.Z
end

local function MakeDrag(frame, handle)
    local d, ds, sp = false
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            d = true; ds = i.Position; sp = frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if d and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local dl = i.Position - ds
            frame.Position = UDim2.new(sp.X.Scale, sp.X.Offset + dl.X, sp.Y.Scale, sp.Y.Offset + dl.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then d = false end
    end)
end

local tbDrag, tbDragStart, tbStartPos, tbMoved = false, nil, nil, false
local TB_THRESH = 6

ToggleBtn.InputBegan:Connect(function(inp)
    if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
        tbDrag = true; tbMoved = false
        tbDragStart = inp.Position; tbStartPos = ToggleBtn.Position
        Tw(ToggleBtn, 0.1, { Size = UDim2.fromOffset(50, 50) })
    end
end)

UserInputService.InputChanged:Connect(function(inp)
    if not tbDrag then return end
    if inp.UserInputType ~= Enum.UserInputType.MouseMovement and inp.UserInputType ~= Enum.UserInputType.Touch then return end
    local d = inp.Position - tbDragStart
    if d.Magnitude >= TB_THRESH then tbMoved = true end
    if tbMoved then
        local vp = workspace.CurrentCamera.ViewportSize
        local nx = math.clamp(tbStartPos.X.Offset + d.X, 0, vp.X - 54)
        local ny = math.clamp(tbStartPos.Y.Offset + d.Y, 0, vp.Y - 54)
        ToggleBtn.Position = UDim2.new(tbStartPos.X.Scale, nx, tbStartPos.Y.Scale, ny)
    end
end)

UserInputService.InputEnded:Connect(function(inp)
    if inp.UserInputType ~= Enum.UserInputType.MouseButton1 and inp.UserInputType ~= Enum.UserInputType.Touch then return end
    if not tbDrag then return end
    tbDrag = false
    Tw(ToggleBtn, 0.15, { Size = UDim2.fromOffset(44, 44) })
    if not tbMoved then
        Main.Visible = not Main.Visible
        Tw(ToggleBtn, 0.12, { BackgroundColor3 = Main.Visible and C.PurpleDm or C.Card })
    end
end)

local function OpenTab(name)
    if CurTab then
        Pages[CurTab].Visible = false
        Tw(TabBtns[CurTab], 0.12, { BackgroundColor3 = C.Card2, TextColor3 = C.Sub })
    end
    CurTab = name
    Pages[name].Visible = true
    Tw(TabBtns[name], 0.12, { BackgroundColor3 = C.Purple, TextColor3 = C.White })
end

local function SliderRow(page, label, min, max, default, fmt, onChange)
    local val = default

    local Row = Instance.new("Frame")
    Row.Size = UDim2.new(1, 0, 0, 62)
    Row.BackgroundColor3 = C.Card; Row.Parent = Pages[page]
    Rnd(Row, 8); Str(Row, C.Stroke, 1)

    local L = Lbl(Row, label, 12, C.Text)
    L.Position = UDim2.fromOffset(12, 7); L.Size = UDim2.new(1, -70, 0, 16)

    local VL = Lbl(Row, (fmt or "%d"):format(val), 11, C.Purple, Enum.TextXAlignment.Right)
    VL.Position = UDim2.new(1, -58, 0, 7); VL.Size = UDim2.fromOffset(48, 16)

    local TBG = Instance.new("Frame")
    TBG.Size = UDim2.new(1, -24, 0, 5); TBG.Position = UDim2.fromOffset(12, 38)
    TBG.BackgroundColor3 = C.Card2; TBG.Parent = Row; Rnd(TBG, 4)

    local Fill = Instance.new("Frame")
    Fill.BackgroundColor3 = C.Purple
    Fill.Size = UDim2.new((val - min) / (max - min), 0, 1, 0); Fill.Parent = TBG; Rnd(Fill, 4)

    local Knob = Instance.new("Frame")
    Knob.Size = UDim2.fromOffset(11, 11); Knob.AnchorPoint = Vector2.new(0.5, 0.5)
    Knob.Position = UDim2.new((val - min) / (max - min), 0, 0.5, 0)
    Knob.BackgroundColor3 = C.White; Knob.Parent = TBG; Rnd(Knob, 20)

    local drg = false
    local function Update(px)
        local pct = math.clamp((px - TBG.AbsolutePosition.X) / TBG.AbsoluteSize.X, 0, 1)
        val = math.floor(min + (max - min) * pct)
        Fill.Size = UDim2.new(pct, 0, 1, 0); Knob.Position = UDim2.new(pct, 0, 0.5, 0)
        VL.Text = (fmt or "%d"):format(val)
        if onChange then onChange(val) end
    end
    TBG.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then drg = true; Update(i.Position.X) end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if drg and i.UserInputType == Enum.UserInputType.MouseMovement then Update(i.Position.X) end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then drg = false end
    end)
end

local function GetRole(player)
    local char = player.Character
    local bag  = player:FindFirstChildOfClass("Backpack")

    local toolList = {}
    if char then
        for _, t in pairs(char:GetChildren()) do
            if t:IsA("Tool") then table.insert(toolList, t) end
        end
    end
    if bag then
        for _, t in pairs(bag:GetChildren()) do
            if t:IsA("Tool") then table.insert(toolList, t) end
        end
    end

    for _, tool in pairs(toolList) do
        local n = tool.Name:lower()
        if n:find("knife") or n == "ghostify" or n:find("blade") then
            return "murderer"
        end
        if n == "sheriff" or n:find("revolver") or n:find("gun") or n:find("pistol") then
            return "sheriff"
        end
    end

    local attr = player:GetAttribute("Role") or player:GetAttribute("role")
    if attr then
        local a = tostring(attr):lower()
        if a:find("murd") then return "murderer" end
        if a:find("sheriff") then return "sheriff" end
    end

    if not char then return "innocent" end
    return "innocent"
end

local function GetMyRole()
    local r = GetRole(LP)
    return r, ROLE_COLOR[r] or C.Green
end

--// ============================================================
--// GET AIMBOT TARGET
--// ============================================================

local function GetTarget()
    local center  = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local bestBone, bestDist = nil, Cfg.AimbotFOV
    local count = 0

    for _, pl in pairs(Players:GetPlayers()) do
        if pl == LP then continue end
        local char = pl.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        if not root or not hum or hum.Health <= 0 then continue end

        local head = char:FindFirstChild("Head") or root
        local s2D, onScreen = W2V(head.Position)
        if not onScreen then continue end

        if Cfg.AimbotVisCheck then
            local rp = RaycastParams.new()
            rp.FilterType = Enum.RaycastFilterType.Exclude
            rp.FilterDescendantsInstances = { LP.Character }
            local res = workspace:Raycast(Camera.CFrame.Position, (head.Position - Camera.CFrame.Position).Unit * 2000, rp)
            if not res or not res.Instance:IsDescendantOf(char) then continue end
        end

        local d = (s2D - center).Magnitude
        if d < bestDist then bestDist = d; bestBone = head end
        count += 1
    end

    ActiveBox.Set("targets", count, C.Text)
    return bestBone
end

local function GetOrMakeESP(pl)

    return t
end

local function GetBoxBounds(char)
    local root = char:FindFirstChild("HumanoidRootPart")
    local head = char:FindFirstChild("Head")
    if not root then return end
    local topPos = head and (head.Position + Vector3.new(0, head.Size.Y * 0.5 + 0.2, 0)) or (root.Position + Vector3.new(0, 3.5, 0))
    local botPos = root.Position - Vector3.new(0, 3.2, 0)
    local top2D, topOn = W2V(topPos)
    local bot2D = W2V(botPos)
    if not topOn or (Camera.CFrame.Position - root.Position).Magnitude > 800 then return end
    local h = math.abs(bot2D.Y - top2D.Y)
    if h < 2 then return end
    local w = h * 0.46
    return top2D, w, h, root
end

local isTweeningCoin = false

task.spawn(function()
    while Gui and Gui.Parent do
        task.wait(0.2)

        if Cfg.AutoCoinOn and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = LP.Character.HumanoidRootPart
            local hum = LP.Character:FindFirstChildOfClass("Humanoid")
            
            if hum and hum.Health > 0 and not isTweeningCoin then
                local targetCoin = nil
                local minDist = math.huge

                -- ค้นหาเหรียญตามโครงสร้างที่คุณระบุ (รองรับทั้งระบุ Hotel และสแกนหา CoinContainer ทั่ว Workspace)
                for _, obj in pairs(workspace:GetDescendants()) do
                    if obj.Name == "Coin_Server" or obj.Name:lower():find("coin") then
                        if obj:IsA("BasePart") or (obj:IsA("Model") and obj.PrimaryPart) then
                            local pos = obj.Position or obj.PrimaryPart.Position
                            local dist = (hrp.Position - pos).Magnitude
                            if dist < minDist then
                                minDist = dist
                                targetCoin = obj
                            end
                        end
                    end
                end

                if targetCoin and targetCoin.Parent then
                    isTweeningCoin = true
                    local originalNoclip = Cfg.NoclipOn
                    Cfg.NoclipOn = true -- เปิด Noclip อัตโนมัติขณะ Tween เก็บเหรียญ

                    local cPos = targetCoin.Position or (targetCoin.PrimaryPart and targetCoin.PrimaryPart.Position)
                    if cPos then
                        local dist = (hrp.Position - cPos).Magnitude
                        local duration = math.clamp(dist / Cfg.CoinSpeed, 0.1, 4)
                        
                        local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear)
                        currentCoinTween = TweenService:Create(hrp, tweenInfo, {CFrame = CFrame.new(cPos + Vector3.new(0, 3, 0))})
                        currentCoinTween:Play()
                        
                        -- รอจนกว่าจะบินไปถึงเหรียญเสร็จสิ้น
                        local elapsed = 0
                        while elapsed < duration and targetCoin and targetCoin.Parent and (Cfg.AutoCoinOn or isTweeningCoin) do
                            local dt = task.wait(0.05)
                            elapsed += dt
                            -- เช็คอัปเดตตำแหน่งเหรียญกรณีเหรียญขยับ
                            if targetCoin and targetCoin.Parent then
                                cPos = targetCoin.Position or (targetCoin.PrimaryPart and targetCoin.PrimaryPart.Position) or cPos
                            end
                        end
                    end

                    -- คืนค่า Noclip กลับตามเดิม (ถ้าผู้เล่นไม่ได้เปิด Noclip ค้างไว้ตั้งแต่แรก)
                    if not originalNoclip then
                        Cfg.NoclipOn = false
                    end
                    isTweeningCoin = false
                    currentCoinTween = nil
                end
            end
        end
        
        -- กรณีผู้เล่นกดปิดฟังก์ชันกลางคัน แต่ยังอยู่ในระหว่าง Tween ให้เคลียร์สถานะเมื่อบินถึงเป้าหมาย
        if not Cfg.AutoCoinOn and isTweeningCoin then
            -- ปล่อยให้มันรัน Tween ตัวปัจจุบันจนจบ แล้วลูปจะตัดจบเองเพราะ Cfg.AutoCoinOn เป็น false
        end
    end
end)

local InvisCache = {}
local InvisConn = nil

function ApplyInvis(enable)
    local char = LP.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")

    if enable then
        InvisCache = {}
        for _, obj in pairs(char:GetDescendants()) do
            pcall(function()
                if obj:IsA("BasePart") then
                    InvisCache[obj] = { lt = obj.LocalTransparencyModifier }
                    obj.LocalTransparencyModifier = 1
                end
                if obj:IsA("Decal") or obj:IsA("SurfaceAppearance") then
                    InvisCache[obj] = { t = obj.Transparency }
                    obj.Transparency = 1
                end
            end)
        end
        if hrp then hrp.LocalTransparencyModifier = 1 end
    else
        for obj, saved in pairs(InvisCache) do
            pcall(function()
                if obj:IsA("BasePart") and saved.lt ~= nil then
                    obj.LocalTransparencyModifier = saved.lt
                end
                if (obj:IsA("Decal") or obj:IsA("SurfaceAppearance")) and saved.t ~= nil then
                    obj.Transparency = saved.t
                end
            end)
        end
        InvisCache = {}
    end
end
